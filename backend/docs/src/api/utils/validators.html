 34class FieldValidator:
 35    """Chainable field validator."""
 36
 37    def __init__(self, field_name: str, value: Any):
 38        self.field_name = field_name
 39        self.value = value
 40        self.errors: List[ValidationError] = []
 41        self._stopped = False
 42
 43    def required(self) -> 'FieldValidator':
 44        """Check if field is present and not None."""
 45        if not self._stopped and self.value is None:
 46            self.errors.append(ValidationError(self.field_name, f"{self.field_name} is required", "required"))
 47            self._stopped = True
 48        return self
 49
 50    def optional(self) -> 'FieldValidator':
 51        """Mark field as optional - stop validation chain if None."""
 52        if self.value is None:
 53            self._stopped = True
 54        return self
 55
 56    def is_type(self, expected_type: type, type_name: str = None) -> 'FieldValidator':
 57        """Validate value is of expected type."""
 58        if self._stopped:
 59            return self
 60        type_name = type_name or expected_type.__name__
 61        if not isinstance(self.value, expected_type):
 62            self.errors.append(ValidationError(
 63                self.field_name,
 64                f"{self.field_name} must be a {type_name}",
 65                "type_error"
 66            ))
 67            self._stopped = True
 68        return self
 69
 70    def in_range(self, min_val: Optional[float] = None, max_val: Optional[float] = None) -> 'FieldValidator':
 71        """Validate numeric value is within range."""
 72        if self._stopped:
 73            return self
 74        if min_val is not None and self.value < min_val:
 75            self.errors.append(ValidationError(
 76                self.field_name,
 77                f"{self.field_name} must be at least {min_val}",
 78                "min_value"
 79            ))
 80        elif max_val is not None and self.value > max_val:
 81            self.errors.append(ValidationError(
 82                self.field_name,
 83                f"{self.field_name} must be at most {max_val}",
 84                "max_value"
 85            ))
 86        return self
 87
 88    def in_set(self, allowed_values: set, case_insensitive: bool = False) -> 'FieldValidator':
 89        """Validate value is in a set of allowed values."""
 90        if self._stopped:
 91            return self
 92        
 93        check_value = self.value
 94        check_set = allowed_values
 95        
 96        if case_insensitive and isinstance(self.value, str):
 97            check_value = self.value.lower()
 98            check_set = {v.lower() if isinstance(v, str) else v for v in allowed_values}
 99        
100        if check_value not in check_set:
101            allowed_list = ', '.join(str(v) for v in sorted(allowed_values))
102            self.errors.append(ValidationError(
103                self.field_name,
104                f"{self.field_name} must be one of: {allowed_list}",
105                "not_in_set"
106            ))
107            self._stopped = True
108        
109        return self
110
111    def transform(self, transformer: Callable[[Any], Any], error_message: str = None) -> 'FieldValidator':
112        """Transform value using provided function."""
113        if self._stopped:
114            return self
115        try:
116            self.value = transformer(self.value)
117        except (ValueError, TypeError) as e:
118            # Log the original exception since it's replaced by a user-friendly message
119            logger.debug(
120                f"Transform failed for '{self.field_name}': "
121                f"{type(e).__name__}: {e} (raw value: {self.value!r})"
122            )
123            message = error_message or f"Invalid {self.field_name} format"
124            self.errors.append(ValidationError(self.field_name, message, "transform_error"))
125            self._stopped = True
126        return self
127
128    def is_valid(self) -> bool:
129        return len(self.errors) == 0
130
131    def get_result(self) -> ValidationResult:
132        if self.errors:
133            return ValidationResult.failure(
134                self.errors[0].field,
135                self.errors[0].message,
136                self.errors[0].code
137            )
138        return ValidationResult.success(self.value)
139
140    def as_bool(self) -> 'FieldValidator':
141        """Parse field as boolean."""
142        if self._stopped:
143            return self
144        self.value = parse_bool_from_string(self.value)
145        return self
146
147    def strip_string(self) -> 'FieldValidator':
148        """Strip whitespace from string values."""
149        if self._stopped or self.value is None:
150            return self
151        if isinstance(self.value, str):
152            self.value = self.value.strip()
153            if not self.value:
154                self.errors.append(ValidationError(
155                    self.field_name,
156                    f"{self.field_name} cannot be empty",
157                    "empty_string"
158                ))
159                self._stopped = True
160        return self

163class RequestValidator:
164    """Validate request data with fluent interface."""
165
166    def __init__(self, data: Dict[str, Any]):
167        self.data = data or {}
168        self.validated: Dict[str, Any] = {}
169        self.errors: List[ValidationError] = []
170
171    def field(self, field_name: str) -> FieldValidator:
172        """Start validating a field."""
173        return FieldValidator(field_name, self.data.get(field_name))
174
175    def validate_field(self, field_name: str, validator: FieldValidator) -> 'RequestValidator':
176        """Add validated field to results."""
177        result = validator.get_result()
178        if result.is_valid:
179            if result.value is not None:
180                self.validated[field_name] = result.value
181        else:
182            self.errors.append(result.error)
183        return self
184
185    def is_valid(self) -> bool:
186        return len(self.errors) == 0
187
188    def get_errors(self) -> List[Dict[str, str]]:
189        return [{"field": e.field, "message": e.message, "code": e.code} for e in self.errors]
190
191    def first_error_message(self) -> Optional[str]:
192        return self.errors[0].message if self.errors else None

250    """Parse boolean from string representation."""
251    if value is None:

352def validate_single_field(
353    field_name: str,
354    value: Any,
355    required: bool = False,
356    field_type: type = None,
357    transformer: Callable = None,
358    min_val: Optional[float] = None,
359    max_val: Optional[float] = None,
360    error_message: str = None
361) -> ValidationResult:
362    """
363    Validate a single field with common options.
364    
365    Simplifies validation for single fields without needing RequestValidator.
366    """
367    validator = FieldValidator(field_name, value)
368
369    if required:
370        validator.required()
371    else:
372        validator.optional()
373
374    if validator._stopped:
375        return validator.get_result()
376
377    if field_type:
378        validator.is_type(field_type)
379
380    if transformer:
381        validator.transform(transformer, error_message)
382
383    if min_val is not None or max_val is not None:
384        validator.in_range(min_val, max_val)
385
386    return validator.get_result()
