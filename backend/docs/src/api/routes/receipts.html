 43def process_receipt_images(file_path: str) -> Optional[Receipt]:
 44    """Load and process receipt image(s), returning the best result."""
 45    logger.debug(f"Loading receipt images from: {file_path}")
 46    receipts = receipt_loader.process_files(file_path, yield_pages=True)
 47
 48    if not receipts:
 49        logger.warning(f"No receipt images loaded from: {file_path}")
 50        return None
 51
 52    if len(receipts) > 1:
 53        logger.debug(f"Multi-page document detected: {len(receipts)} pages")
 54        processed_receipts = [receipt_extractor.process_receipt(r) for r in receipts]
 55        best = max(processed_receipts, key=lambda r: r.confidence)
 56        logger.debug(
 57            f"Selected best page with confidence: {best.confidence} "
 58            f"from {len(processed_receipts)} pages"
 59        )
 60        return best
 61
 62    processed = receipt_extractor.process_receipt(receipts[0])
 63    logger.debug(f"Single page processed with confidence: {processed.confidence}")
 64    return processed

 67def apply_form_overrides(receipt: Receipt, form) -> Receipt:
 68    """
 69    Apply form data overrides to a receipt.
 70    Raises AppError on invalid input.
 71    """
 72    overridden_fields = []
 73
 74    if form.get('vendor'):
 75        receipt.vendor = form['vendor']
 76        overridden_fields.append('vendor')
 77
 78    if form.get('amount'):
 79        try:
 80            receipt.amount = float(form['amount'])
 81            overridden_fields.append('amount')
 82        except ValueError:
 83            logger.warning(f"Invalid amount format: {form['amount']}")
 84            raise invalid_value('Invalid amount format', field='amount')
 85
 86    if form.get('date'):
 87        try:
 88            receipt.date = datetime.fromisoformat(form['date'])
 89            overridden_fields.append('date')
 90        except ValueError:
 91            logger.warning(f"Invalid date format: {form['date']}")
 92            raise invalid_value('Invalid date format. Use YYYY-MM-DD', field='date')
 93
 94    if overridden_fields:
 95        logger.debug(f"Applied form overrides: {overridden_fields}")
 96
 97    return receipt

100def validate_receipt_filters(args) -> Dict:
101    """
102    Parse and validate receipt filter parameters from request args.
103    Raises AppError on invalid input.
104    """
105    logger.debug(f"Validating receipt filters: {list(args.keys())}")
106
107    is_valid, pagination, error = validate_pagination(args)
108    if not is_valid:
109        logger.warning(f"Pagination validation failed: {error}")
110        raise invalid_value(error)
111
112    is_valid, date_filters, error = validate_date_range_filters(args)
113    if not is_valid:
114        logger.warning(f"Date range validation failed: {error}")
115        raise invalid_value(error)
116
117    validator = RequestValidator(args)
118    validator.validated.update(date_filters)
119
120    add_string_filters(validator.validated, args, ['vendor'])
121
122    validator.validate_field('min_confidence',
123        validator.field('min_confidence').optional().transform(parse_int).in_range(0, 3))
124
125    if validator.validated.get('start_date') and validator.validated.get('end_date'):
126        if validator.validated['start_date'] > validator.validated['end_date']:
127            logger.warning("start_date is after end_date")
128            raise invalid_value('start_date cannot be after end_date', field='start_date')
129
130    if not validator.is_valid():
131        logger.warning(f"Filter validation failed: {validator.first_error_message()}")
132        raise invalid_value(validator.first_error_message())
133
134    return {**pagination, **validator.validated}

137def validate_receipt_update(data: Dict) -> Dict:
138    """
139    Validate and extract receipt update fields from request data.
140    Raises AppError on invalid input.
141    """
142    logger.debug(f"Validating update data with keys: {list(data.keys())}")
143
144    validator = RequestValidator(data)

146    add_string_filters(validator.validated, data, ['vendor', 'raw_text'])
147
148    if 'amount' in data:

151                validator.field('amount').transform(parse_float, 'Invalid amount format'))
152        else:

154
155    if 'date' in data:
156        if data['date'] is not None:
157            validator.validate_field('date',
158                validator.field('date').transform(parse_date, 'Invalid date format. Use YYYY-MM-DD'))
159        else:
160            validator.validated['date'] = None
161
162    if 'confidence' in data:
163        if data['confidence'] is not None:
164            validator.validate_field('confidence',
165                validator.field('confidence').transform(parse_int).in_range(0, 3))
166        else:
167            validator.validated['confidence'] = None
168
169    if not validator.is_valid():
170        logger.warning(f"Validation failed: {validator.first_error_message()}")
171        raise invalid_value(validator.first_error_message())

176def validate_confirm_receipt(data: Dict) -> Dict:
177    """
178    Validate receipt confirmation data.
179    Raises AppError on invalid input.
180    """
181    logger.debug(f"Validating confirmation data with keys: {list(data.keys())}")
182
183    error = require_at_least_one(data, ['original_filename', 'vendor'])
184    if error:
185        logger.warning(f"Validation failed: {error}")
186        raise invalid_value(error)
187
188    validator = RequestValidator(data)
189
190    validator.validate_field('original_filename',
191        validator.field('original_filename').required().strip_string())
192    validator.validate_field('vendor',
193        validator.field('vendor').required().strip_string())
194
195    if 'amount' in data and data['amount'] is not None:
196        validator.validate_field('amount',
197            validator.field('amount').transform(parse_float, 'Invalid amount format'))
198
199    if 'date' in data and data['date']:
200        validator.validate_field('date',
201            validator.field('date').transform(parse_date, 'Invalid date format. Use YYYY-MM-DD'))
202
203    validator.validate_field('confidence',
204        validator.field('confidence').transform(parse_int).in_range(0, 3))
205    if 'confidence' not in validator.validated:
206        validator.validated['confidence'] = 0
207
208    for field_name in ['id', 'file_path', 'stored_filename', 'raw_text']:
209        if field_name in data:
210            validator.validated[field_name] = data[field_name]
211
212    if not validator.is_valid():
213        logger.warning(f"Validation failed: {validator.first_error_message()}")
214        raise invalid_value(validator.first_error_message())
215
216    return validator.validated

226@bp.route('/receipts/upload-stream', methods=['POST'])
227@log_route(logger)
228def upload_receipts_stream():
229    """Upload and process multiple receipt images with streaming responses."""
230    temp_files = []
231
232    try:
233        file_handler = FileHandler.from_app_config(prefix="receipt")
234        files = request.files.getlist('files')
235
236        if not files or (len(files) == 1 and files[0].filename == ''):
237            logger.warning("No files provided in stream upload")
238            return create_error_sse_response("No files provided")
239
240        logger.info(f"Received {len(files)} files for stream processing")
241
242        for file in files:
243            validation = file_handler.validate_file(file)
244            if not validation.is_valid:
245                logger.warning(f"Skipping invalid file: {validation.error}")
246                continue
247
248            fd, temp_path = tempfile.mkstemp(suffix=validation.extension)
249            try:
250                with os.fdopen(fd, 'wb') as f:
251                    file.save(f)
252                temp_files.append((validation.secured_filename, temp_path))
253                logger.debug(f"Saved {validation.secured_filename} to temp: {temp_path}")
254            except Exception as e:
255                if os.path.exists(temp_path):
256                    os.unlink(temp_path)
257                logger.error(f"Failed to save {file.filename}: {e}")
258
259        if not temp_files:
260            logger.warning("No valid files after validation")
261            return create_error_sse_response("No valid files to process")
262
263        logger.info(f"Processing {len(temp_files)} valid files via stream")
264
265        form_data = request.form.to_dict() if request.form else None
266
267        processor = ReceiptStreamProcessor(
268            upload_folder=file_handler.ensure_upload_folder(),
269            allowed_extensions=file_handler.allowed_extensions
270        )
271
272        def cleanup_and_stream():
273            try:
274                yield from processor.process_files(temp_files, form_data)
275            finally:
276                for _, temp_path in temp_files:
277                    if os.path.exists(temp_path):
278                        try:
279                            os.unlink(temp_path)
280                            logger.debug(f"Cleaned up temp file: {temp_path}")
281                        except Exception as e:
282                            logger.error(f"Failed to cleanup {temp_path}: {e}")
283
284        return create_sse_response(
285            cleanup_and_stream(),
286            headers={"Access-Control-Allow-Origin": "*"}
287        )
288
289    except Exception as e:
290        logger.error(f"Stream upload failed: {e}", exc_info=True)
291        for _, temp_path in temp_files:
292            if os.path.exists(temp_path):
293                try:
294                    os.unlink(temp_path)
295                except Exception:
296                    pass
297        return create_error_sse_response("Internal server error", str(e))

304@bp.route('/receipts/upload', methods=['POST'])
305@handle_errors(entity='Receipt')
306@log_route(logger)
307def upload_receipt():
308    """
309    Upload a receipt image file, process it, and save to database.
310
311    Expected: multipart/form-data with 'file' field
312    Optional form fields:
313        - vendor: Override extracted vendor
314        - amount: Override extracted amount
315        - date: Override extracted date (YYYY-MM-DD)
316    """
317    file_handler = FileHandler.from_app_config(prefix="receipt")
318
319    if 'file' not in request.files:
320        logger.warning("No file field in request")
321        raise required('file')
322
323    file = request.files['file']
324    validation = file_handler.validate_file(file)
325
326    if not validation.is_valid:
327        logger.warning(f"File validation failed: {validation.error}")
328        raise invalid_value(validation.error, field='file')
329
330    with TempFileManager() as temp_manager:
331        temp_path = temp_manager.save_file_to_temp(file, suffix=validation.extension)
332
333        logger.info(f"Processing uploaded file: {validation.secured_filename}")
334
335        receipt = process_receipt_images(str(temp_path))
336
337        if receipt is None:
338            logger.warning(f"Failed to process image: {validation.secured_filename}")
339            raise AppError(
340                code=ErrorCode.INVALID_VALUE,
341                message='Unable to process the uploaded image',
342                status_code=422,
343                field='file',
344            )
345
346        # Raises AppError on bad overrides
347        receipt = apply_form_overrides(receipt, request.form)
348
349        stored_filename = file_handler.generate_stored_filename(validation.secured_filename)
350        stored_path = file_handler.move_to_permanent(temp_path, stored_filename)
351        logger.debug(f"Moved to permanent storage: {stored_path}")
352
353        receipt.original_filename = validation.secured_filename
354        receipt.stored_filename = stored_filename
355        receipt.file_path = stored_path
356
357        receipt_id = receipt_repository.save(receipt)
358
359        if receipt_id is None:
360            file_handler.delete_file(stored_path)
361            logger.error(f"Database save failed, cleaned up file: {stored_path}")
362            raise AppError(
363                code=ErrorCode.DATABASE_ERROR,
364                message='Failed to save receipt',
365                status_code=500,
366            )
367
368        saved_receipt = receipt_repository.get_by_id(receipt_id)
369
370        logger.info(
371            f"Uploaded receipt {receipt_id}: {validation.secured_filename} "
372            f"| confidence={receipt.confidence}"
373        )
374
375        return success_response(
376            {
377                'receipt': ReceiptFormatter.summary(saved_receipt),
378                'message': 'Receipt uploaded and processed successfully'
379            },
380            status_code=201
381        )

388@bp.route('/receipts', methods=['GET'])
389@handle_errors(entity='Receipt')

391def get_receipts():
392    """Get list of receipts with optional filters and pagination."""
393    filters = validate_receipt_filters(request.args)
394
395    limit = filters.pop('limit')
396    offset = filters.pop('offset')
397
398    active_filters = {k: v for k, v in filters.items() if v is not None}
399    if active_filters:
400        logger.debug(f"Active filters: {active_filters}")
401
402    receipts = receipt_repository.get_all(limit=limit, offset=offset, **filters)
403    formatted_receipts = ReceiptFormatter.summary_list(receipts)
404
405    response = {
406        'receipts': formatted_receipts,
407        'pagination': {
408            'limit': limit,
409            'offset': offset,
410            'count': len(formatted_receipts),
411            'has_more': len(formatted_receipts) == limit
412        },
413        'filters': {
414            'vendor': filters.get('vendor'),
415            'min_confidence': filters.get('min_confidence'),
416            'start_date': request.args.get('start_date'),
417            'end_date': request.args.get('end_date')
418        }
419    }
420
421    logger.info(
422        f"Retrieved {len(formatted_receipts)} receipts "
423        f"(offset={offset}, limit={limit})"

556@bp.route('/receipts/<int:receipt_id>/reprocess', methods=['POST'])

559def reprocess_receipt(receipt_id: int):
560    """Rerun OCR processing on an existing receipt."""
561    receipt = receipt_repository.get_by_id(receipt_id)
562    if receipt is None:
563        raise not_found('Receipt', receipt_id)
564
565    file_path = receipt.get('file_path')
566
567    if not file_path:
568        logger.warning(f"No image file associated with receipt {receipt_id}")
569        raise invalid_value(
570            f'Receipt {receipt_id} has no associated image file to reprocess'
571        )
572
573    path = Path(file_path)
574
575    if not path.exists():
576        logger.error(f"Image file missing from disk: {file_path}")
577        raise AppError(
578            code=ErrorCode.NOT_FOUND,
579            message=f'Image file not found on disk for receipt {receipt_id}',
580            status_code=404,
581            entity='Receipt image',
582        )
583
584    data = request.get_json(silent=True) or {}
585    keep_overrides = data.get('keep_overrides', False)
586
587    original_vendor = receipt.get('vendor') if keep_overrides else None
588    original_amount = receipt.get('amount') if keep_overrides else None
589    original_date = receipt.get('date') if keep_overrides else None
590
591    logger.info(f"Reprocessing receipt {receipt_id} | keep_overrides={keep_overrides}")
592
593    processed = process_receipt_images(str(path))
594
595    if processed is None:
596        logger.warning(f"Reprocessing failed for receipt {receipt_id}")
597        raise AppError(
598            code=ErrorCode.INVALID_VALUE,
599            message='Unable to reprocess the image',
600            status_code=422,
601        )
602
603    update_kwargs = {
604        'confidence': processed.confidence,
605        'raw_text': processed.extracted_text,
606    }
607
608    if keep_overrides:
609        preserved = []
610        if original_vendor is not None:
611            preserved.append('vendor')
612        else:
613            update_kwargs['vendor'] = processed.vendor
614        if original_amount is not None:
615            preserved.append('amount')
616        else:
617            update_kwargs['amount'] = processed.amount
618        if original_date is not None:
619            preserved.append('date')
620        else:
621            update_kwargs['date'] = processed.date
622        if preserved:
623            logger.debug(f"Preserved override fields: {preserved}")
624    else:
625        update_kwargs['vendor'] = processed.vendor
626        update_kwargs['amount'] = processed.amount
627        update_kwargs['date'] = processed.date
628
629    updated_receipt = receipt_repository.update(receipt_id, **update_kwargs)
630
631    logger.info(
632        f"Reprocessed receipt {receipt_id} "
633        f"| new_confidence={processed.confidence}"
634    )
635
636    return success_response({
637        'receipt': ReceiptFormatter.summary(updated_receipt),
638        'message': 'Receipt reprocessed successfully',
639        'reprocessing': {
640            'kept_overrides': keep_overrides,
641            'new_confidence': processed.confidence,
642            'extracted_vendor': processed.vendor,
643            'extracted_amount': processed.amount,
644            'extracted_date': processed.date.isoformat() if processed.date else None
645        }
646    })

653def process_receipt():
654    """Run OCR on receipt image without saving to database."""
655    file_handler = FileHandler.from_app_config(prefix="receipt")
656
657    if 'file' not in request.files:
658        logger.warning("No file field in request")
659        raise required('file')
660
661    file = request.files['file']
662    validation = file_handler.validate_file(file)
663
664    if not validation.is_valid:
665        logger.warning(f"File validation failed: {validation.error}")
666        raise invalid_value(validation.error, field='file')
667
668    with TempFileManager() as temp_manager:
669        temp_path = temp_manager.save_file_to_temp(file, suffix=validation.extension)
670
671        logger.info(f"Processing image (no save): {validation.secured_filename}")
672
673        receipts = receipt_loader.process_files(str(temp_path), yield_pages=True)
674
675        if not receipts:
676            logger.warning(f"Failed to process image: {validation.secured_filename}")
677            raise AppError(
678                code=ErrorCode.INVALID_VALUE,
679                message='Unable to process the uploaded image',
680                status_code=422,
681                field='file',
682            )
683
684        results = []
685        for receipt in receipts:
686            processed = receipt_extractor.process_receipt(receipt)
687            results.append(ReceiptFormatter.extracted_data(processed))
688
689        logger.debug(f"Processed {len(results)} pages from {validation.secured_filename}")
690
691        if len(results) == 1:
692            response = {
693                'success': True,
694                'original_filename': validation.secured_filename,
695                'extracted_data': results[0],
696                'page_count': 1
697            }
698        else:

701            response = {

705                'all_pages': results,

713            f"amount={response['extracted_data'].get('amount')}"
