src.models.files

Data models for tabular file operations.

Defines the data classes and enums used throughout the file-import pipeline — from initial validation through preview to final import. Each stage produces a typed result object that captures the outcome, metadata, and any errors or warnings.

Typical flow:
  1. Validate file → ValidationResult (with ColumnInfo per column)
  2. Preview rows → PreviewResult
  3. Import data → ImportResult

All result classes support to_dict() and to_json() for API serialization.

 78@dataclass
 79class ColumnInfo:
 80    """Metadata about a single column in a tabular file.
 81
 82    Built during validation and included in `ValidationResult.columns`.
 83    Provides enough information for the UI to let users select or rename
 84    columns before import.
 85
 86    Attributes:
 87        name: Column header text.
 88        index: Zero-based positional index in the source file.
 89        data_type: Detected type as a string (see `DataType`).
 90        nullable: Whether the column contains any null values.
 91        sample_values: A handful of representative non-null values.
 92        null_count: Total number of null/missing cells.
 93        unique_count: Number of distinct non-null values.
 94    """
 95    name: str
 96    index: int
 97    data_type: str
 98    nullable: bool
 99    sample_values: List[Any] = field(default_factory=list)
100    null_count: int = 0
101    unique_count: int = 0
102
103    def __post_init__(self):
104        """Coerce numpy scalar types to native Python types.
105
106        Pandas operations often return numpy types (`np.bool_`,
107        `np.int64`, etc.) which don't serialize cleanly to JSON. This
108        converts them at construction time so downstream code doesn't
109        need to worry about it.
110        """
111        # Convert nullable to Python bool
112        if isinstance(self.nullable, (np.bool_, np.generic)):
113            self.nullable = bool(self.nullable)
114
115        # Convert counts to Python int
116        if isinstance(self.null_count, (np.integer, np.generic)):
117            self.null_count = int(self.null_count)
118
119        if isinstance(self.unique_count, (np.integer, np.generic)):
120            self.unique_count = int(self.unique_count)
121
122        # Convert index to Python int
123        if isinstance(self.index, (np.integer, np.generic)):
124            self.index = int(self.index)
125
126    def to_dict(self) -> Dict[str, Any]:
127        """Return a plain dict representation suitable for JSON serialization."""
128        return asdict(self)

131@dataclass
132class ValidationResult:
133    """Outcome of validating an uploaded tabular file.
134
135    Captures file metadata, detected format properties, per-column info,
136    and any errors or warnings. Used by the preview step and returned to
137    the UI so the user can confirm before importing.
138
139    Attributes:
140        is_valid: True if the file can be imported (no blocking errors).
141        file_path: Absolute path to the stored file on disk.
142        file_name: Original filename as uploaded by the user.
143        file_type: Detected `FileType` value (as a string).
144        file_size_bytes: File size in bytes.
145        row_count: Number of data rows (excluding header).
146        column_count: Number of columns detected.
147        columns: Per-column metadata (see `ColumnInfo`).
148        has_header: Whether a header row was detected.
149        detected_delimiter: Delimiter character for delimited text files.
150            None for binary formats.
151        detected_encoding: Character encoding (e.g. "utf-8").
152        errors: Blocking issues that prevent import.
153        warnings: Non-blocking issues the user should be aware of.
154        validated_at: ISO 8601 timestamp of when validation ran.
155    """
156    is_valid: bool
157    file_path: str
158    file_name: str
159    file_type: str
160    file_size_bytes: int
161    row_count: int
162    column_count: int
163    columns: List[ColumnInfo]
164    has_header: bool
165    detected_delimiter: Optional[str]
166    detected_encoding: str
167    errors: List[str] = field(default_factory=list)
168    warnings: List[str] = field(default_factory=list)
169    validated_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
170
171    def to_dict(self) -> Dict[str, Any]:
172        """Convert to a plain dict, ensuring nested `ColumnInfo` objects are expanded."""
173        result = asdict(self)
174        result['columns'] = [col.to_dict() if isinstance(col, ColumnInfo) else col
175                           for col in self.columns]
176        return result
177
178    def to_json(self) -> str:
179        """Serialize to a formatted JSON string.
180
181        Uses `default=str` to handle any remaining non-serializable
182        types (e.g. `Path` objects).
183        """
184        return json.dumps(self.to_dict(), default=str, indent=2)

233    """Outcome of importing tabular data into the database.
234

252        data: The imported rows as dicts.
253        upload_id: FK to the `uploads` table row, if the data was
254            persisted.
255        errors: Blocking issues encountered during import.
256        warnings: Non-blocking issues or per-row warnings. May contain
257            strings or structured dicts.
258        imported_at: ISO 8601 timestamp.
259    """
260    success: bool
261    file_name: str
262    file_type: str
263    start_row: int
264    rows_imported: int

267    columns_requested: Optional[List[Any]]
268    column_mapping: Optional[Dict[str, str]]
269    data: List[Dict[str, Any]]
270    upload_id: Optional[int] = None
271    errors: List[str] = field(default_factory=list)

273    imported_at: str = field(default_factory=lambda: datetime.now(timezone.utc).isoformat())
