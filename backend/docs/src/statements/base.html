src.statements.base

Statement processing: base classes and configuration dataclasses.

Defines the contract for turning a bank's CSV/Excel export into a normalized transaction DataFrame. Each bank format is described declaratively by a StatementConfig; the StatementProcessor pipeline consumes that config.

Two ways to support a bank:
  • Config only (common) — write a StatementConfig in configs/<bank>.py, the registry wraps it in ConfigurableStatementProcessor. No subclassing.
  • Custom processor (rare) — subclass StatementProcessor and override individual pipeline steps. Needed when the export requires row-level transformations config can't express (e.g. Revolut).

Pipeline overview — see StatementProcessor.process_statement(): raw rows → skip header/footer → validate columns → filter junk → parse dates → parse amounts → extract description → defaults → categorizer (assign party) → stamp ids → enforce output shape

 43@dataclass
 44class AmountConfig:
 45    """
 46    How to extract a signed amount + credit/debit flag from source columns.
 47
 48    Banks represent amounts one of three ways. Configure **exactly one**:
 49
 50      1. **Split columns** — separate Credit and Debit columns, each positive.
 51         → set `credit_column` and `debit_column`
 52
 53      2. **Signed column** — one column, sign indicates direction
 54         (negative = debit).
 55         → set `amount_column` and `signed_amount=True`
 56
 57      3. **Amount + indicator** — one unsigned column plus a "CR"/"DR" flag.
 58         → set `amount_column`, `credit_indicator_column`,
 59           `credit_indicator_value`
 60
 61    Output convention regardless of input: debits are negative, credits
 62    positive (`debit_is_negative=True`). Flip that if your downstream
 63    expects the opposite.
 64    """
 65    # Option 1: Separate credit/debit columns
 66    credit_column: Optional[str] = None
 67    debit_column: Optional[str] = None
 68    
 69    # Option 2: Single amount column with sign or indicator
 70    amount_column: Optional[str] = None
 71    credit_indicator_column: Optional[str] = None
 72    credit_indicator_value: Optional[str] = None
 73    
 74    # Option 3: Single amount column, negative = debit
 75    signed_amount: bool = False
 76    
 77    # Common settings
 78    currency_symbols: list[str] = field(default_factory=lambda: ['€', '$', '£'])
 79    decimal_separator: str = '.'
 80    thousands_separator: str = ','
 81    debit_is_negative: bool = True  # Convention: debits stored as negative

 92@dataclass
 93class StatementConfig:
 94    """
 95    Declarative description of one bank's statement export format.
 96
 97    One instance per (bank, account type) combination, living in
 98    `configs/<bank>.py`. Tells the processor which columns to read,
 99    how to parse them, and which rows to skip.
100
101    Minimal example::
102
103        StatementConfig(
104            bank_name="AIB",
105            account_type="Current Account",
106            date_config=DateConfig(column="Date", format="%d/%m/%Y"),
107            amount_config=AmountConfig(
108                credit_column="Credit",
109                debit_column="Debit",
110            ),
111            description_column="Description",
112        )
113
114    Attributes:
115        bank_name: Human-readable bank name ("AIB", "Revolut").
116        account_type: Variant ("Current Account", "Credit Card").
117            Combined with bank_name to form `display_name`.
118        date_config: Which column holds the date and how to parse it.
119        amount_config: Which column(s) hold the amount — see `AmountConfig`
120            for the three supported shapes.
121        description_column: Source column with the transaction narrative.
122        balance_column: Optional running balance. Not used in processing;
123            reserved for future reconciliation.
124        reference_column: Optional reference/memo column.
125        skip_rows_start: Leading rows to discard (title rows, logos).
126        skip_rows_end: Trailing rows to discard (totals, disclaimers).
127        exclude_patterns: Regexes matched against the description column.
128            Matching rows are dropped — use for "OPENING BALANCE" lines etc.
129        defaults: Overrides for standard output fields, applied to every
130            row from this config. E.g. `{"is_kids": True}` for payments related
131            to children.
132    """
133    bank_name: str
134    account_type: str
135
136    date_config: DateConfig
137    amount_config: AmountConfig
138    description_column: str
139
140    balance_column: Optional[str] = None
141    reference_column: Optional[str] = None
142
143    skip_rows_start: int = 0
144    skip_rows_end: int = 0
145
146    exclude_patterns: list[str] = field(default_factory=list)
147    defaults: dict[str, Any] = field(default_factory=dict)
148
149    def __post_init__(self):
150        """Validate configuration."""
151        amt = self.amount_config
152        has_split = amt.credit_column and amt.debit_column
153        has_single = amt.amount_column is not None
154
155        if not has_split and not has_single:
156            raise ValueError(
157                "AmountConfig must specify either credit_column/debit_column "
158                "or amount_column"
159            )
160        
161    @property
162    def required_columns(self) -> list[str]:
163        """All source columns this config expects to find in the input data."""
164        cols = [self.date_config.column, self.description_column]
165
166        amt = self.amount_config
167        if amt.credit_column and amt.debit_column:
168            cols.extend([amt.credit_column, amt.debit_column])
169        elif amt.amount_column:
170            cols.append(amt.amount_column)
171            if amt.credit_indicator_column:
172                cols.append(amt.credit_indicator_column)
173
174        # Dedupe while preserving order
175        seen = set()
176        return [c for c in cols if c and not (c in seen or seen.add(c))]
177
178    @property
179    def display_name(self) -> str:
180        return f"{self.bank_name} {self.account_type}"

192class StatementProcessor(ABC):
193    """
194    Abstract pipeline for parsing statement rows into normalized transactions.
195
196    The pipeline lives in `process_statement()`. Each stage is a separate
197    private method, so subclasses can override one step without
198    reimplementing the whole thing.
199
200    **You usually don't subclass this.** If the bank's format fits a
201    `StatementConfig`, the registry uses `ConfigurableStatementProcessor`
202    automatically. Subclass only for banks that need transformations
203    config can't express — row merging, multi-currency normalization, etc.
204
205    Instance state:
206        account_id: Stamped onto every output row.
207        upload_id: Stamped onto every output row.
208        categorizer: Assigns `party_id` and `confidence` per description.
209            Injectable for testing; defaults to a fresh TransactionCategorizer.
210        transactions: Output DataFrame — populated by `process_statement()`.
211        warnings: Non-fatal issues (e.g. some dates unparseable) collected
212            during processing. The caller should surface these to the user.
213
214    Class attributes:
215        OUTPUT_COLUMNS: Exact column set the repository layer expects.
216            `_finalize_columns()` guarantees this shape.
217        _DATE_FORMAT_FALLBACKS: Formats tried when the configured date
218            format fails. Ordered roughly by likelihood for Irish/EU banks.
219    """
220    
221    # Standard output columns expected by the database
222    OUTPUT_COLUMNS = [
223        'transaction_date',
224        'amount', 
225        'description',
226        'is_credit',
227        'is_kids',
228        'is_one_off',
229        'receipt_id',
230        'cleaned_description',
231        'party_id',
232        'confidence',
233        'account_id',
234        'upload_id'
235    ]
236
237    _DATE_FORMAT_FALLBACKS = [
238        '%Y-%m-%dT%H:%M:%SZ',    # 2022-10-13T01:02:03Z   ← the format in your logs
239        '%Y-%m-%dT%H:%M:%S%z',   # 2022-10-13T01:02:03+00:00
240        '%Y-%m-%dT%H:%M:%S',     # 2022-10-13T01:02:03
241        '%Y-%m-%d',              # 2022-10-13
242        '%d/%m/%Y', '%d/%m/%y',  # European
243        '%m/%d/%Y', '%m/%d/%y',  # US
244        '%d-%m-%Y', '%d.%m.%Y',
245        '%Y/%m/%d',
246    ]
247
248    def __init__(
249        self,
250        account_id: int,
251        upload_id: int,
252        categorizer: Optional[TransactionCategorizer] = None,
253    ):
254        self.account_id = account_id
255        self.upload_id = upload_id
256        self.categorizer = categorizer or TransactionCategorizer()
257        self.transactions: Optional[pd.DataFrame] = None
258        self.warnings: list[ProcessingWarning] = []
259
260        logger.debug(
261            f"Initialized {self.__class__.__name__}: "
262            f"account_id={account_id}, upload_id={upload_id}"
263        )

271    def process_statement(self, transactions: list[dict]) -> pd.DataFrame:
272        """
273        Run the full pipeline on raw statement rows.
274
275        Args:
276            transactions: Rows as dicts, keys = source file column headers.
277                Typically produced by `utils/tabular_files/readers.py`.
278
279        Returns:
280            DataFrame with exactly `OUTPUT_COLUMNS`, one row per transaction.
281            Also stored on `self.transactions`.
282
283        Raises:
284            AppError (INVALID_FORMAT): Required columns are missing, or no
285                dates are parseable by any known format. The error's `details`
286                dict is structured for direct display — the frontend's
287                ColumnMismatchPanel consumes it.
288
289        Side effects:
290            Appends `ProcessingWarning` entries to `self.warnings` for
291            partial failures (e.g. 5 of 200 dates unparseable → those rows
292            are dropped, a warning is recorded). Caller must surface these.
293        """
294        logger.info(
295            f"Processing {self.config.display_name} statement: "
296            f"{len(transactions)} rows for account {self.account_id}"
297        )
298
299        df = pd.DataFrame(transactions)
300
301        # Skip configured rows BEFORE validating columns — headers/footers
302        # won't affect the column set, but doing it first matches the
303        # original behaviour order.
304        if self.config.skip_rows_start > 0:
305            df = df.iloc[self.config.skip_rows_start:]
306        if self.config.skip_rows_end > 0:
307            df = df.iloc[:-self.config.skip_rows_end]
308
309        # Validate columns before any processing touches them
310        self._validate_required_columns(df)
311
312        df = self._filter_rows(df)
313        df = self._parse_dates(df)
314        df = self._parse_amounts(df)
315        df = self._parse_description(df)
316        df = self._apply_defaults(df)
317        df = self._categorize(df)
318
319        df['account_id'] = int(self.account_id)
320        df['upload_id'] = int(self.upload_id)
321
322        self.transactions = self._finalize_columns(df)
323
324        logger.info(
325            f"Processing complete: {len(self.transactions)} transactions"
326        )
327        return self.transactions

329    def _validate_required_columns(self, df: pd.DataFrame) -> None:
330        """
331        Verify all columns required by the config exist in the input data.
332
333        Raises AppError with details about what's missing and what was found,
334        so the user can tell whether they picked the wrong account or the
335        wrong file.
336        """
337        required = self.config.required_columns
338        actual = list(df.columns)
339        missing = [c for c in required if c not in actual]
340
341        if not missing:
342            logger.debug(
343                f"Column check passed for {self.config.display_name}: "
344                f"all {len(required)} required columns present"
345            )
346            return
347
348        logger.warning(
349            f"Column mismatch for {self.config.display_name}: "
350            f"missing={missing}, found={actual}"
351        )
352
353        raise AppError(
354            code=ErrorCode.INVALID_FORMAT,
355            message=(
356                f"This file doesn't match the '{self.config.display_name}' "
357                f"statement format. "
358                f"Missing column(s): {', '.join(missing)}. "
359                f"File contains: {', '.join(actual) or '(no columns)'}."
360            ),
361            status_code=422,
362            entity='Statement',
363            details={
364                'statement_format': self.config.display_name,
365                'account_id': self.account_id,
366                'missing_columns': missing,
367                'required_columns': required,
368                'found_columns': actual,
369            },
370        )

372    def _filter_rows(self, df: pd.DataFrame) -> pd.DataFrame:
373        """Remove non-transaction rows based on configured patterns."""
374        initial_count = len(df)
375        
376        for pattern in self.config.exclude_patterns:
377            mask = df[self.config.description_column].str.contains(
378                pattern, case=False, na=False, regex=True
379            )
380            df = df[~mask]
381        
382        filtered = initial_count - len(df)
383        if filtered > 0:
384            logger.debug(f"Filtered out {filtered} non-transaction rows")
385            
386        return df

388    def _parse_dates(self, df: pd.DataFrame) -> pd.DataFrame:
389        date_cfg = self.config.date_config
390        col = date_cfg.column
391        rows_before = len(df)
392
393        sample_values = df[col].dropna().astype(str).head(3).tolist()
394        logger.debug(
395            f"Parsing dates: column={col!r}, format={date_cfg.format!r}, "
396            f"dayfirst={date_cfg.dayfirst} | samples: {sample_values}"
397        )
398
399        parsed, used_format = self._attempt_date_parse(df[col], date_cfg)
400        df['transaction_date'] = parsed
401
402        cfg_label = date_cfg.format or f'auto(dayfirst={date_cfg.dayfirst})'
403        if used_format != cfg_label:
404            logger.warning(
405                f"Configured format {date_cfg.format!r} didn't match this file. "
406                f"Used {used_format!r} instead."
407            )
408
409        invalid_mask = df['transaction_date'].isna()
410        invalid_count = int(invalid_mask.sum())
411
412        if invalid_count > 0:
413            invalid_samples = (
414                df.loc[invalid_mask, col].dropna().astype(str).head(5).tolist()
415            )
416            logger.warning(
417                f"Dropping {invalid_count}/{rows_before} rows with unparseable "
418                f"dates (best format: {used_format!r}) | samples: {invalid_samples}"
419            )
420
421            # ── This is the bit that was missing ──
422            self.warnings.append(ProcessingWarning(
423                code='DATES_UNPARSEABLE',
424                message=(
425                    f"{invalid_count} of {rows_before} rows were skipped because "
426                    f"the date in column '{col}' couldn't be parsed."
427                ),
428                details={
429                    'dropped': invalid_count,
430                    'total': rows_before,
431                    'column': col,
432                    'format_used': used_format,
433                    'sample_values': invalid_samples,
434                },
435            ))
436
437            df = df.dropna(subset=['transaction_date'])
438
439        return df

442    def _attempt_date_parse(
443        self,
444        series: pd.Series,
445        date_cfg: DateConfig,
446    ) -> tuple[pd.Series, str]:
447        """
448        Best-effort date parsing across multiple format candidates.
449
450        Tries the configured format, then every fallback, and keeps whichever
451        parses the **most** rows. Only short-circuits on 100% — a format that
452        parses 80% might still lose to one that parses 95%, so we try them all.
453
454        Why this is lenient: banks change export formats without warning, and
455        users often open files in Excel first which silently reformats dates.
456        A fuzzy parse + warning beats failing the whole import.
457
458        Returns:
459            Tuple of (parsed series, label of winning format). The label is
460            for logging and for the `ProcessingWarning` shown to the user.
461
462        Raises:
463            AppError (INVALID_FORMAT): Zero rows parsed by any candidate.
464        """
465        total = int(series.notna().sum())
466        if total == 0:
467            return pd.to_datetime(series, errors='coerce'), 'n/a (empty)'
468
469        # Build an ordered list of (label, to_datetime kwargs) to try.
470        cfg_label = date_cfg.format or f'auto(dayfirst={date_cfg.dayfirst})'
471        attempts: list[tuple[str, dict]] = [
472            (cfg_label, {'format': date_cfg.format, 'dayfirst': date_cfg.dayfirst}),
473        ]
474
475        # dayfirst=True mangles ISO-8601 when auto-detecting — try without it too.
476        if date_cfg.format is None and date_cfg.dayfirst:
477            attempts.append(('auto(dayfirst=False)', {'format': None, 'dayfirst': False}))
478
479        # pandas >= 2.0 has a flexible ISO8601 parser — the try/except
480        # in the loop below handles older versions gracefully.
481        attempts.append(('ISO8601', {'format': 'ISO8601'}))
482
483        best_fmt = date_cfg.format or 'auto-detected'
484
485        for fmt in self._DATE_FORMAT_FALLBACKS:
486            if fmt != date_cfg.format:
487                attempts.append((fmt, {'format': fmt}))
488
489        best_parsed: pd.Series | None = None
490        best_valid = -1
491        best_label = cfg_label
492
493        for label, kwargs in attempts:
494            try:
495                parsed = pd.to_datetime(series, errors='coerce', **kwargs)
496            except (ValueError, TypeError):
497                continue
498
499            valid = int(parsed.notna().sum())
500            logger.debug(f"Date format {label!r}: {valid}/{total} parsed")
501
502            if valid > best_valid:
503                best_parsed, best_valid, best_label = parsed, valid, label
504
505            if valid == total:
506                return parsed, label          # perfect match — done
507
508        if best_valid <= 0:
509            sample = series.dropna().astype(str).head(3).tolist()
510            raise AppError(
511                code=ErrorCode.INVALID_FORMAT,
512                message=(
513                    f"Could not parse any dates in column '{date_cfg.column}'. "
514                    f"Sample values: {sample}."
515                ),
516                status_code=422,
517                entity='Statement',
518                field=date_cfg.column,
519                details={
520                    'column': date_cfg.column,
521                    'sample_values': sample,
522                    'formats_tried': [lbl for lbl, _ in attempts],
523                },
524            )
525
526        return best_parsed, best_label

528    def _parse_amounts(self, df: pd.DataFrame) -> pd.DataFrame:
529        """Parse transaction amounts and determine credit/debit."""
530        amt_cfg = self.config.amount_config
531        
532        if amt_cfg.credit_column and amt_cfg.debit_column:
533            df = self._parse_split_amounts(df, amt_cfg)
534        elif amt_cfg.signed_amount:
535            df = self._parse_signed_amount(df, amt_cfg)
536        else:
537            df = self._parse_single_amount(df, amt_cfg)
538        
539        return df
540
541    def _parse_split_amounts(
542        self, df: pd.DataFrame, cfg: AmountConfig
543    ) -> pd.DataFrame:
544        """Handle statements with separate credit/debit columns."""
545        credit = df[cfg.credit_column].apply(self._clean_amount)
546        debit = df[cfg.debit_column].apply(self._clean_amount)
547        
548        if cfg.debit_is_negative:
549            df['amount'] = credit - debit
550        else:
551            df['amount'] = credit + debit
552            
553        df['is_credit'] = credit > 0
554        return df
555
556    def _parse_signed_amount(
557        self, df: pd.DataFrame, cfg: AmountConfig
558    ) -> pd.DataFrame:
559        """Handle statements with a single signed amount column."""
560        df['amount'] = df[cfg.amount_column].apply(self._clean_amount)
561        df['is_credit'] = df['amount'] > 0
562        return df
563
564    def _parse_single_amount(
565        self, df: pd.DataFrame, cfg: AmountConfig
566    ) -> pd.DataFrame:
567        """Handle statements with amount + credit indicator."""
568        df['amount'] = df[cfg.amount_column].apply(self._clean_amount)
569        
570        if cfg.credit_indicator_column:
571            df['is_credit'] = (
572                df[cfg.credit_indicator_column] == cfg.credit_indicator_value
573            )
574            # Apply sign based on credit/debit
575            if cfg.debit_is_negative:
576                df.loc[~df['is_credit'], 'amount'] *= -1
577        else:
578            # Assume all positive, user must handle sign elsewhere
579            df['is_credit'] = True
580            
581        return df
582
583    def _clean_amount(self, value: Any) -> float:
584        """Convert a currency string to float."""
585        if pd.isna(value):
586            return 0.0
587            
588        if isinstance(value, (int, float)):
589            return float(value)
590            
591        s = str(value).strip()
592        
593        # Remove currency symbols
594        for symbol in self.config.amount_config.currency_symbols:
595            s = s.replace(symbol, '')
596        
597        # Handle thousands/decimal separators
598        s = s.replace(self.config.amount_config.thousands_separator, '')
599        s = s.replace(self.config.amount_config.decimal_separator, '.')
600        s = s.replace(' ', '')
601        
602        if s in ('', '-', 'N/A', 'n/a'):
603            return 0.0
604            
605        return float(s)
606
607    def _parse_description(self, df: pd.DataFrame) -> pd.DataFrame:
608        """Extract and clean description."""
609        df['description'] = (
610            df[self.config.description_column]
611            .astype(str)
612            .fillna('')
613            .str.strip()
614        )
615        return df
616
617    def _apply_defaults(self, df: pd.DataFrame) -> pd.DataFrame:
618        """Apply default values for standard fields."""
619        defaults = {
620            'is_kids': False,
621            'is_one_off': False,
622            'receipt_id': None,
623            **self.config.defaults
624        }
625        
626        for col, default in defaults.items():
627            if col not in df.columns:
628                df[col] = default
629

272    def process_statement(self, transactions: list[dict]) -> pd.DataFrame:
273        """
274        Run the full pipeline on raw statement rows.
275
276        Args:
277            transactions: Rows as dicts, keys = source file column headers.
278                Typically produced by `utils/tabular_files/readers.py`.
279

281            DataFrame with exactly `OUTPUT_COLUMNS`, one row per transaction.
282            Also stored on `self.transactions`.

286                dates are parseable by any known format. The error's `details`
287                dict is structured for direct display — the frontend's
288                ColumnMismatchPanel consumes it.
289

291            Appends `ProcessingWarning` entries to `self.warnings` for
292            partial failures (e.g. 5 of 200 dates unparseable → those rows
293            are dropped, a warning is recorded). Caller must surface these.
294        """
295        logger.info(
296            f"Processing {self.config.display_name} statement: "
297            f"{len(transactions)} rows for account {self.account_id}"
298        )
299
300        df = pd.DataFrame(transactions)
301
302        # Skip configured rows BEFORE validating columns — headers/footers
303        # won't affect the column set, but doing it first matches the
304        # original behaviour order.
305        if self.config.skip_rows_start > 0:

307        if self.config.skip_rows_end > 0:

309

319

321        df['upload_id'] = int(self.upload_id)
322
323        self.transactions = self._finalize_columns(df)
324
325        logger.info(
326            f"Processing complete: {len(self.transactions)} transactions"

443    def _attempt_date_parse(
444        self,
445        series: pd.Series,
446        date_cfg: DateConfig,
447    ) -> tuple[pd.Series, str]:
448        """
449        Best-effort date parsing across multiple format candidates.
450
451        Tries the configured format, then every fallback, and keeps whichever
452        parses the **most** rows. Only short-circuits on 100% — a format that
453        parses 80% might still lose to one that parses 95%, so we try them all.
454
455        Why this is lenient: banks change export formats without warning, and
456        users often open files in Excel first which silently reformats dates.
457        A fuzzy parse + warning beats failing the whole import.
458
459        Returns:
460            Tuple of (parsed series, label of winning format). The label is
461            for logging and for the `ProcessingWarning` shown to the user.
462
463        Raises:
464            AppError (INVALID_FORMAT): Zero rows parsed by any candidate.
465        """
466        total = int(series.notna().sum())
467        if total == 0:
468            return pd.to_datetime(series, errors='coerce'), 'n/a (empty)'
469
470        # Build an ordered list of (label, to_datetime kwargs) to try.
471        cfg_label = date_cfg.format or f'auto(dayfirst={date_cfg.dayfirst})'
472        attempts: list[tuple[str, dict]] = [
473            (cfg_label, {'format': date_cfg.format, 'dayfirst': date_cfg.dayfirst}),
474        ]
475
476        # dayfirst=True mangles ISO-8601 when auto-detecting — try without it too.
477        if date_cfg.format is None and date_cfg.dayfirst:
478            attempts.append(('auto(dayfirst=False)', {'format': None, 'dayfirst': False}))
479
480        # pandas >= 2.0 has a flexible ISO8601 parser — the try/except
481        # in the loop below handles older versions gracefully.
482        attempts.append(('ISO8601', {'format': 'ISO8601'}))
483
484        best_fmt = date_cfg.format or 'auto-detected'
485
486        for fmt in self._DATE_FORMAT_FALLBACKS:
487            if fmt != date_cfg.format:
488                attempts.append((fmt, {'format': fmt}))
489
490        best_parsed: pd.Series | None = None
491        best_valid = -1
492        best_label = cfg_label
493
494        for label, kwargs in attempts:
495            try:
496                parsed = pd.to_datetime(series, errors='coerce', **kwargs)
497            except (ValueError, TypeError):
498                continue
499
500            valid = int(parsed.notna().sum())
501            logger.debug(f"Date format {label!r}: {valid}/{total} parsed")
502
503            if valid > best_valid:
504                best_parsed, best_valid, best_label = parsed, valid, label
505
506            if valid == total:
507                return parsed, label          # perfect match — done
508
509        if best_valid <= 0:
510            sample = series.dropna().astype(str).head(3).tolist()
511            raise AppError(
512                code=ErrorCode.INVALID_FORMAT,
513                message=(
514                    f"Could not parse any dates in column '{date_cfg.column}'. "
515                    f"Sample values: {sample}."
516                ),
517                status_code=422,
518                entity='Statement',
519                field=date_cfg.column,
520                details={
521                    'column': date_cfg.column,
522                    'sample_values': sample,
523                    'formats_tried': [lbl for lbl, _ in attempts],
524                },
525            )
526
527        return best_parsed, best_label

531        amt_cfg = self.config.amount_config
532        
533        if amt_cfg.credit_column and amt_cfg.debit_column:

535        elif amt_cfg.signed_amount:

542    def _parse_split_amounts(
543        self, df: pd.DataFrame, cfg: AmountConfig
544    ) -> pd.DataFrame:

546        credit = df[cfg.credit_column].apply(self._clean_amount)
547        debit = df[cfg.debit_column].apply(self._clean_amount)
548        
549        if cfg.debit_is_negative:
550            df['amount'] = credit - debit

565    def _parse_single_amount(
566        self, df: pd.DataFrame, cfg: AmountConfig
567    ) -> pd.DataFrame:
568        """Handle statements with amount + credit indicator."""
569        df['amount'] = df[cfg.amount_column].apply(self._clean_amount)
570        
571        if cfg.credit_indicator_column:
572            df['is_credit'] = (
573                df[cfg.credit_indicator_column] == cfg.credit_indicator_value
574            )
575            # Apply sign based on credit/debit
576            if cfg.debit_is_negative:
577                df.loc[~df['is_credit'], 'amount'] *= -1
578        else:
579            # Assume all positive, user must handle sign elsewhere
580            df['is_credit'] = True
581            
582        return df

584    def _clean_amount(self, value: Any) -> float:
585        """Convert a currency string to float."""
586        if pd.isna(value):
587            return 0.0
588            
589        if isinstance(value, (int, float)):
590            return float(value)
591            
592        s = str(value).strip()
593        
594        # Remove currency symbols
595        for symbol in self.config.amount_config.currency_symbols:
596            s = s.replace(symbol, '')
597        
598        # Handle thousands/decimal separators
599        s = s.replace(self.config.amount_config.thousands_separator, '')
600        s = s.replace(self.config.amount_config.decimal_separator, '.')
601        s = s.replace(' ', '')
602        
603        if s in ('', '-', 'N/A', 'n/a'):
604            return 0.0
605            
606        return float(s)

609        """Extract and clean description."""
610        df['description'] = (
611            df[self.config.description_column]
612            .astype(str)
613            .fillna('')
614            .str.strip()
615        )
616        return df

625        }

Abstract pipeline for parsing statement rows into normalized transactions.

The pipeline lives in process_statement(). Each stage is a separate private method, so subclasses can override one step without reimplementing the whole thing.

You usually don't subclass this. If the bank's format fits a StatementConfig, the registry uses ConfigurableStatementProcessor automatically. Subclass only for banks that need transformations config can't express — row merging, multi-currency normalization, etc.

Instance state:

account_id: Stamped onto every output row. upload_id: Stamped onto every output row. categorizer: Assigns party_id and confidence per description. Injectable for testing; defaults to a fresh TransactionCategorizer. transactions: Output DataFrame — populated by process_statement(). warnings: Non-fatal issues (e.g. some dates unparseable) collected during processing. The caller should surface these to the user.

Class attributes:

OUTPUT_COLUMNS: Exact column set the repository layer expects. _finalize_columns() guarantees this shape. _DATE_FORMAT_FALLBACKS: Formats tried when the configured date format fails. Ordered roughly by likelihood for Irish/EU banks.

272    def process_statement(self, transactions: list[dict]) -> pd.DataFrame:
273        """
274        Run the full pipeline on raw statement rows.
275
276        Args:
277            transactions: Rows as dicts, keys = source file column headers.
278                Typically produced by `utils/tabular_files/readers.py`.
279
280        Returns:
281            DataFrame with exactly `OUTPUT_COLUMNS`, one row per transaction.
282            Also stored on `self.transactions`.
283
284        Raises:
285            AppError (INVALID_FORMAT): Required columns are missing, or no
286                dates are parseable by any known format. The error's `details`
287                dict is structured for direct display — the frontend's
288                ColumnMismatchPanel consumes it.
289
290        Side effects:
291            Appends `ProcessingWarning` entries to `self.warnings` for
292            partial failures (e.g. 5 of 200 dates unparseable → those rows
293            are dropped, a warning is recorded). Caller must surface these.
294        """
295        logger.info(
296            f"Processing {self.config.display_name} statement: "
297            f"{len(transactions)} rows for account {self.account_id}"
298        )
299
300        df = pd.DataFrame(transactions)
301
302        # Skip configured rows BEFORE validating columns — headers/footers
303        # won't affect the column set, but doing it first matches the
304        # original behaviour order.
305        if self.config.skip_rows_start > 0:
306            df = df.iloc[self.config.skip_rows_start:]
307        if self.config.skip_rows_end > 0:
308            df = df.iloc[:-self.config.skip_rows_end]
309
310        # Validate columns before any processing touches them
311        self._validate_required_columns(df)
312
313        df = self._filter_rows(df)
314        df = self._parse_dates(df)
315        df = self._parse_amounts(df)
316        df = self._parse_description(df)
317        df = self._apply_defaults(df)
318        df = self._categorize(df)
319
320        df['account_id'] = int(self.account_id)
321        df['upload_id'] = int(self.upload_id)
322
323        self.transactions = self._finalize_columns(df)
324
325        logger.info(
326            f"Processing complete: {len(self.transactions)} transactions"
327        )
328        return self.transactions
