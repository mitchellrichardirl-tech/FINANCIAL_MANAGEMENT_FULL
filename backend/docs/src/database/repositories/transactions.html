src.database.repositories.transactions

Repository for transaction data access.

Provides the TransactionRepository class, which handles CRUD, bulk operations, filtered queries, and receipt-linking against the transactions table.

Return types vary by method:
  • Write operations and simple reads return Transaction model instances.
  • Hierarchy-aware reads (anything that joins through parties → types → sub_categories → categories) return plain dicts, since the extra columns don't fit the Transaction model.

 72                        is_credit, is_kids, is_one_off, account_id,
 73                        upload_id, party_id, receipt_id)
 74                    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
 75                ''', (

 78                    transaction.is_credit, transaction.is_kids,

 92    def bulk_add_transactions(self, df: pd.DataFrame) -> List[int]:
 93        """Bulk insert transactions from a pandas DataFrame.
 94
 95        Validates that required columns are present, fills missing
 96        optional columns with defaults, normalises dates to ISO format,
 97        and inserts all rows in a single transaction.
 98
 99        Args:
100            df: DataFrame with at least `transaction_date`, `amount`,
101                and `description` columns. Optional columns
102                (`cleaned_description`, `is_credit`, `is_kids`,
103                `is_one_off`, `account_id`, `upload_id`, `party_id`,
104                `receipt_id`) default to None/False if absent.
105
106        Returns:
107            List of auto-generated `id` values for the inserted rows.
108
109        Raises:
110            ValueError: If any required columns are missing.
111            DatabaseError: On any database failure (entire batch is
112                rolled back).
113
114        Note:
115            IDs are inferred from `MAX(id)` after insert, which assumes
116            no concurrent inserts to the same table.
117        """
118        logger.info(f"Bulk inserting {len(df)} transactions from DataFrame")
119
120        try:
121            required_columns = ['transaction_date', 'amount', 'description']
122            missing_columns = [col for col in required_columns if col not in df.columns]
123            if missing_columns:
124                logger.warning(f"Missing required columns: {missing_columns}")
125                raise ValueError(f"Missing required columns: {missing_columns}")
126
127            df_copy = df.copy()
128            df_copy['transaction_date'] = pd.to_datetime(
129                df_copy['transaction_date']
130            ).dt.strftime('%Y-%m-%d')
131
132            optional_defaults = {
133                'cleaned_description': None,
134                'is_credit': False,
135                'is_kids': False,
136                'is_one_off': False,
137                'account_id': None,
138                'upload_id': None,
139                'party_id': None,
140                'receipt_id': None
141            }
142
143            for col, default in optional_defaults.items():
144                if col not in df_copy.columns:
145                    df_copy[col] = default
146
147            columns_order = [
148                'transaction_date', 'amount', 'description', 'cleaned_description',
149                'is_credit', 'is_kids', 'is_one_off', 'account_id',
150                'upload_id', 'party_id', 'receipt_id'
151            ]
152
153            data = [tuple(row) for row in df_copy[columns_order].values]
154
155            with self.db.transaction() as conn:
156                cursor = conn.cursor()
157                cursor.executemany('''
158                    INSERT INTO transactions 
159                    (transaction_date, amount, description, cleaned_description,
160                        is_credit, is_kids, is_one_off, account_id,
161                        upload_id, party_id, receipt_id)
162                    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
163                ''', data)
164
165                cursor.execute('SELECT MAX(id) FROM transactions')
166                max_id = cursor.fetchone()[0]
167                transaction_ids = list(range(max_id - len(data) + 1, max_id + 1))
168
169            logger.info(f"Bulk inserted {len(data)} transactions (ids {transaction_ids[0]}-{transaction_ids[-1]})")
170            return transaction_ids
171
172        except ValueError:
173            raise
174        except Exception as e:
175            logger.error(f"Failed to bulk add transactions: {e}")
176            raise DatabaseError(f"Failed to bulk add transactions: {e}") from e

239    def get_transactions(self, limit: Optional[int] = None) -> List[Transaction]:
240        """Fetch transactions as `Transaction` model instances.
241
242        Joins to `accounts`, `parties`, and `receipts` to populate
243        `account_name`, `related_party_name`, and `receipt_filename`.
244        Ordered by date descending (most recent first).
245
246        Args:
247            limit: Maximum rows to return. None for all.
248
249        Returns:
250            List of `Transaction` instances. Empty list if the table
251            is empty.
252
253        Raises:
254            DatabaseError: On query failure.
255        """
256        try:
257            with self.db.get_connection() as conn:
258                cursor = conn.cursor()
259                query = '''
260                    SELECT t.*,
261                        a.account_name as account_name,
262                        p.name as related_party_name,
263                        r.original_filename as receipt_filename
264                    FROM transactions t
265                    LEFT JOIN accounts a ON t.account_id = a.id
266                    LEFT JOIN parties p ON t.party_id = p.id
267                    LEFT JOIN receipts r ON t.receipt_id = r.id
268                    ORDER BY t.transaction_date DESC
269                '''
270                if limit:
271                    query += ' LIMIT ?'
272                    cursor.execute(query, (limit,))
273                else:
274                    cursor.execute(query)
275
276                rows = cursor.fetchall()
277                transactions = [Transaction(**dict(row)) for row in rows]
278
279                logger.debug(f"Retrieved {len(transactions)} transactions")
280                return transactions
281
282        except Exception as e:
283            logger.error(f"Failed to get transactions: {e}")
284            raise DatabaseError(f"Failed to get transactions: {e}") from e

327    def find_receipt_match_candidates(
328        self,
329        amount: float,
330        transaction_date: str,
331        amount_tolerance: float = 0.01,
332        date_tolerance_days: int = 7,
333        include_matched: bool = False
334    ) -> List[Transaction]:
335        """Find transactions that could match a given receipt.
336
337        Used by the receipt-linking workflow. Searches for transactions
338        whose amount and date are within the given tolerances, optionally
339        excluding transactions that already have a receipt linked.
340
341        Results are ordered by date proximity first, then amount
342        proximity, so the best match appears first.
343
344        Args:
345            amount: Receipt amount to match against.
346            transaction_date: Receipt date in YYYY-MM-DD format.
347            amount_tolerance: Maximum absolute difference in amount.
348                Defaults to 0.01.
349            date_tolerance_days: Maximum difference in days between
350                receipt date and transaction date. Defaults to 7.
351            include_matched: If True, include transactions that already
352                have a receipt linked. Defaults to False.
353
354        Returns:
355            List of `Transaction` instances ordered by match quality
356            (closest first). Empty list if no candidates found.
357
358        Raises:
359            DatabaseError: On query failure.
360        """
361        logger.debug(
362            f"Finding receipt match candidates: amount={amount}, "
363            f"date={transaction_date}, tolerance={amount_tolerance}"
364        )
365
366        try:
367            with self.db.get_connection() as conn:
368                cursor = conn.cursor()
369
370                base_query = '''
371                    SELECT t.*,
372                        a.account_name as account_name,
373                        p.name as related_party_name,
374                        r.original_filename as receipt_filename
375                    FROM transactions t
376                    LEFT JOIN accounts a ON t.account_id = a.id
377                    LEFT JOIN parties p ON t.party_id = p.id
378                    LEFT JOIN receipts r ON t.receipt_id = r.id
379                    WHERE ABS(t.amount - ?) <= ?
380                    AND ABS(julianday(t.transaction_date) - julianday(?)) <= ?
381                '''
382
383                if not include_matched:
384                    base_query += ' AND t.receipt_id IS NULL'
385
386                base_query += '''
387                    ORDER BY ABS(julianday(t.transaction_date) - julianday(?)) ASC,
388                            ABS(t.amount - ?) ASC
389                '''
390
391                cursor.execute(base_query, (
392                    amount, amount_tolerance,
393                    transaction_date, date_tolerance_days,
394                    transaction_date, amount
395                ))
396
397                rows = cursor.fetchall()
398                transactions = [Transaction(**dict(row)) for row in rows]
399
400                logger.debug(
401                    f"Found {len(transactions)} candidates for "
402                    f"amount={amount}, date={transaction_date}"
403                )
404                return transactions
405
406        except Exception as e:
407            logger.error(f"Failed to find receipt match candidates: {e}")
408            raise DatabaseError(f"Failed to find receipt match candidates: {e}") from e

410    def get_transactions_with_hierarchy(
411        self,
412        limit: Optional[int] = None,
413        offset: int = 0,
414        start_date: Optional[str] = None,
415        end_date: Optional[str] = None,
416        party_id: Optional[int] = None,
417        account_id: Optional[int] = None,
418        upload_id: Optional[int] = None,
419        description: Optional[str] = None,
420        cleaned_description: Optional[str] = None,
421        is_kids: Optional[bool] = None,
422        is_one_off: Optional[bool] = None,
423        is_credit: Optional[bool] = None,
424        category_id: Optional[int] = None,
425        sub_category_id: Optional[int] = None,
426        type_id: Optional[int] = None,
427        sort_by: Optional[str] = None,
428        sort_dir: Optional[str] = None
429    ) -> List[Dict[str, Any]]:
430        """Fetch transactions with full category hierarchy and optional filters.
431
432        The primary query method for the transaction listing UI. Joins
433        through the full chain (accounts, parties → types →
434        sub_categories → categories, receipts) and supports filtering
435        at every level plus sortable columns.
436
437        All filter arguments are optional and ANDed together. Returns
438        dicts rather than `Transaction` instances because the joined
439        columns don't fit the model.
440
441        Args:
442            limit: Maximum rows to return. None for all.
443            offset: Rows to skip for pagination. Defaults to 0.
444            start_date: Include transactions on or after this date
445                (YYYY-MM-DD).
446            end_date: Include transactions on or before this date
447                (YYYY-MM-DD).
448            party_id: Filter by exact party.
449            account_id: Filter by exact account.
450            upload_id: Filter by import batch.
451            description: Substring match on raw description.
452            cleaned_description: Substring match on cleaned description.
453            is_kids: Filter by kids flag.
454            is_one_off: Filter by one-off flag.
455            is_credit: Filter by credit/debit direction.
456            category_id: Filter by category (via hierarchy join).
457            sub_category_id: Filter by subcategory (via hierarchy join).
458            type_id: Filter by type (via hierarchy join).
459            sort_by: Column name to sort by. Must be in the whitelist
460                (e.g. "transaction_date", "amount", "party_name").
461                Defaults to "transaction_date". Unrecognised values
462                fall back to the default.
463            sort_dir: "asc" or "desc". Defaults to "desc".
464
465        Returns:
466            List of transaction dicts, each including `account_name`,
467            `account_type`, `party_name`, `type_name`,
468            `sub_category_name`, `category_name`, and receipt fields.
469            Empty list if no matches.
470
471        Raises:
472            DatabaseError: On query failure.
473        """
474
475        # Whitelist of allowed sort columns to prevent SQL injection
476        # Keys are the API field names, values are the actual SQL column references
477        SORTABLE_COLUMNS = {
478            'transaction_date': 't.transaction_date',
479            'amount': 't.amount',
480            'description': 't.description',
481            'cleaned_description': 't.cleaned_description',
482            'is_credit': 't.is_credit',
483            'is_kids': 't.is_kids',
484            'is_one_off': 't.is_one_off',
485            'account_name': 'a.account_name',
486            'party_name': 'p.name',
487            'type_name': 'tp.type',
488            'sub_category_name': 'sc.sub_category',
489            'category_name': 'c.category',
490        }
491
492        try:
493            with self.db.get_connection() as conn:
494                cursor = conn.cursor()
495
496                conditions = []
497                params = []
498
499                if start_date:
500                    conditions.append('t.transaction_date >= ?')
501                    params.append(start_date)
502
503                if end_date:
504                    conditions.append('t.transaction_date <= ?')
505                    params.append(end_date)
506
507                if party_id:
508                    conditions.append('t.party_id = ?')
509                    params.append(party_id)
510
511                if account_id:
512                    conditions.append('t.account_id = ?')
513                    params.append(account_id)
514
515                if upload_id:
516                    conditions.append('t.upload_id = ?')
517                    params.append(upload_id)
518
519                if description:
520                    conditions.append('t.description LIKE ?')
521                    params.append(f'%{description}%')
522
523                if cleaned_description:
524                    conditions.append('t.cleaned_description LIKE ?')
525                    params.append(f'%{cleaned_description}%')
526
527                if is_kids is not None:
528                    conditions.append('t.is_kids = ?')
529                    params.append(int(is_kids))
530
531                if is_one_off is not None:
532                    conditions.append('t.is_one_off = ?')
533                    params.append(int(is_one_off))
534
535                if is_credit is not None:
536                    conditions.append('t.is_credit = ?')
537                    params.append(int(is_credit))
538
539                if category_id:
540                    conditions.append('c.id = ?')
541                    params.append(category_id)
542
543                if sub_category_id:
544                    conditions.append('sc.id = ?')
545                    params.append(sub_category_id)
546
547                if type_id:
548                    conditions.append('tp.id = ?')
549                    params.append(type_id)
550
551                if conditions:
552                    logger.debug(f"Querying transactions with {len(conditions)} filters")
553
554                where_clause = f"WHERE {' AND '.join(conditions)}" if conditions else ""
555
556                # Build ORDER BY clause safely
557                sort_column = SORTABLE_COLUMNS.get(sort_by, 't.transaction_date')
558                sort_direction = 'ASC' if sort_dir == 'asc' else 'DESC'
559
560                # Add secondary sort by id for stable ordering
561                order_clause = f"ORDER BY {sort_column} {sort_direction}, t.id DESC"
562
563                query = f'''
564                    SELECT 
565                        t.*,
566                        a.account_name,
567                        a.account_type,
568                        p.id as party_id,
569                        p.name as party_name,
570                        tp.id as type_id,
571                        tp.type as type_name,
572                        sc.id as sub_category_id,
573                        sc.sub_category as sub_category_name,
574                        c.id as category_id,
575                        c.category as category_name,
576                        r.id as receipt_id,
577                        r.original_filename as receipt_filename,
578                        r.vendor as receipt_vendor,
579                        r.amount as receipt_amount,
580                        r.date as receipt_date
581                    FROM transactions t
582                    LEFT JOIN accounts a ON t.account_id = a.id
583                    LEFT JOIN parties p ON t.party_id = p.id
584                    LEFT JOIN types tp ON p.type_id = tp.id
585                    LEFT JOIN sub_categories sc ON tp.sub_category_id = sc.id
586                    LEFT JOIN categories c ON sc.category_id = c.id
587                    LEFT JOIN receipts r ON t.receipt_id = r.id
588                    {where_clause}
589                    {order_clause}
590                '''
591
592                if limit:
593                    query += ' LIMIT ?'
594                    params.append(limit)
595
596                if offset:
597                    query += ' OFFSET ?'
598                    params.append(offset)
599
600                cursor.execute(query, params)
601                rows = cursor.fetchall()
602
603                logger.debug(
604                    f"Retrieved {len(rows)} transactions with hierarchy "
605                    f"(offset={offset}, limit={limit}, sort={sort_by} {sort_dir})"
606                )
607                return [dict(row) for row in rows]
608
609        except Exception as e:
610            logger.error(f"Failed to get transactions with hierarchy: {e}")
611            raise DatabaseError(f"Failed to get transactions: {e}") from e
