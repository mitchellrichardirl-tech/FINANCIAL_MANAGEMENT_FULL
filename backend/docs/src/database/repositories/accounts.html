src.database.repositories.accounts

Repository for account database operations.

Provides CRUD access to the accounts table through the AccountRepository class. Accounts represent bank or card accounts that transactions are imported from.

Each account has a name, type (e.g. "current", "credit"), and an optional statement_format that links to the statement parser registry for automatic file parsing.

Typical usage:

repo = AccountRepository() account = repo.add_account("Main Current", "current", statement_format="aib") all_accounts = repo.get_all_accounts()

 54    def add_account(
 55        self,
 56        account_name: str,
 57        account_type: str,
 58        statement_format: Optional[str] = None,
 59    ) -> Optional[Dict[str, Any]]:
 60        """Create a new account and return the full record.
 61
 62        Args:
 63            account_name: Display name for the account (must be unique).
 64            account_type: Account kind (e.g. "current", "credit", "savings").
 65            statement_format: Optional identifier for the statement parser
 66                to use when importing files for this account (e.g. "aib",
 67                "revolut"). Can be set later via `update_account()`.
 68
 69        Returns:
 70            Dict of the newly created account row, or None if the row
 71            could not be retrieved after insert.
 72
 73        Raises:
 74            DatabaseError: If the account name already exists or the
 75                insert fails for any other reason.
 76        """
 77        logger.debug(f"Adding account: {account_name} ({account_type})")
 78
 79        try:
 80            with self.db.transaction() as conn:
 81                cursor = conn.cursor()
 82
 83                cursor.execute(
 84                    """INSERT INTO accounts
 85                       (account_name, account_type, statement_format)
 86                       VALUES (?, ?, ?)""",
 87                    (account_name, account_type, statement_format),
 88                )
 89                account_id = cursor.lastrowid
 90
 91                cursor.execute("SELECT * FROM accounts WHERE id = ?", (account_id,))
 92                row = cursor.fetchone()
 93
 94            logger.info(f"Added account {account_id}: {account_name}")
 95            return dict(row) if row else None
 96
 97        except sqlite3.IntegrityError as e:
 98            if "unique" in str(e).lower():
 99                logger.warning(f"Duplicate account name: {account_name}")
100                raise DatabaseError(f"Account already exists: {account_name}") from e
101            logger.error(f"Integrity error adding account: {e}")
102            raise DatabaseError(f"Failed to add account: {e}") from e
103        except Exception as e:
104            logger.error(f"Failed to add account: {e}")
105            raise DatabaseError(f"Failed to add account: {e}") from e

291    def update_account(
292        self,
293        account_id: int,
294        account_name: Optional[str] = None,
295        account_type: Optional[str] = None,
296        statement_format: Optional[str] = None,
297    ) -> Optional[Dict[str, Any]]:
298        """Update an account's fields selectively.
299
300        Only the fields passed with non-None values are modified. If no
301        fields are provided the existing record is returned unchanged.
302
303        Args:
304            account_id: The ID of the account to update.
305            account_name: New display name (must remain unique).
306            account_type: New account type.
307            statement_format: New statement parser identifier.
308
309        Returns:
310            Dict of the updated account row, or None if no account
311            exists with the given ID.
312
313        Raises:
314            DatabaseError: If the new name conflicts with an existing
315                account or the update fails for any other reason.
316        """
317        try:
318            with self.db.transaction() as conn:
319                cursor = conn.cursor()
320
321                updates = []
322                params = []
323                updated_fields = []
324
325                if account_name is not None:
326                    updates.append("account_name = ?")
327                    params.append(account_name)
328                    updated_fields.append("account_name")
329
330                if account_type is not None:
331                    updates.append("account_type = ?")
332                    params.append(account_type)
333                    updated_fields.append("account_type")
334
335                if statement_format is not None:
336                    updates.append("statement_format = ?")
337                    params.append(statement_format)
338                    updated_fields.append("statement_format")
339
340                if not updates:
341                    logger.debug(f"No fields to update for account {account_id}")
342                    return self.get_account_by_id(account_id)
343
344                params.append(account_id)
345                query = f"UPDATE accounts SET {', '.join(updates)} WHERE id = ?"
346                cursor.execute(query, params)
347
348                if cursor.rowcount == 0:
349                    logger.debug(f"Account {account_id} not found for update")
350                    return None
351
352            logger.info(f"Updated account {account_id}: {updated_fields}")
353            return self.get_account_by_id(account_id)
354
355        except sqlite3.IntegrityError as e:
356            if "unique" in str(e).lower():
357                logger.warning(f"Duplicate account name on update: {account_name}")
358                raise DatabaseError(
359                    f"Account name already exists: {account_name}"
360                ) from e
361            logger.error(f"Integrity error updating account {account_id}: {e}")
362            raise DatabaseError(f"Failed to update account: {e}") from e
363        except Exception as e:
364            logger.error(f"Failed to update account {account_id}: {e}")
365            raise DatabaseError(f"Failed to update account: {e}") from e

369    def delete_account(self, account_id: int) -> bool:
370        """Delete an account by ID.
371
372        Checks for associated transactions before deleting. If the
373        account has any linked transactions the delete is refused to
374        prevent orphaned records.
375
376        Args:
377            account_id: The ID of the account to delete.
378
379        Returns:
380            True if the account was deleted, False if no account exists
381            with the given ID.
382
383        Raises:
384            DatabaseError: If the account has associated transactions or
385                the delete fails for any other reason.
386        """
387        try:
388            with self.db.transaction() as conn:
389                cursor = conn.cursor()
390
391                cursor.execute(
392                    "SELECT COUNT(*) as count FROM transactions WHERE account_id = ?",
393                    (account_id,),
394                )
395                count = cursor.fetchone()["count"]
396
397                if count > 0:
398                    logger.warning(
399                        f"Cannot delete account {account_id}: "
400                        f"has {count} associated transactions"
401                    )
402                    raise DatabaseError(
403                        f"Cannot delete account {account_id}: "
404                        f"has {count} associated transaction(s)"
405                    )
406
407                cursor.execute("DELETE FROM accounts WHERE id = ?", (account_id,))
408
409                if cursor.rowcount == 0:
410                    logger.debug(f"Account {account_id} not found for deletion")
411                    return False
412
413                logger.info(f"Deleted account {account_id}")
414                return True
415
416        except DatabaseError:
417            raise
418        except Exception as e:
419            logger.error(f"Failed to delete account {account_id}: {e}")
420            raise DatabaseError(f"Failed to delete account: {e}") from e
