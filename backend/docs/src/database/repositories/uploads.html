src.database.repositories.uploads

Repository for file upload and upload-data access.

Provides the UploadRepository class, which manages two related tables:

- `uploads` — metadata about uploaded files (filename, type, shape).
- `upload_data` — the individual rows from each uploaded file, stored
  as JSON blobs for schema-agnostic access.

Both tables are covered by a single repository because they're always accessed together: an upload is meaningless without its data, and data rows always belong to exactly one upload.

The columns field on uploads and row_data on upload_data are stored as JSON and automatically serialized/deserialized by the repository.

 67        file_type: str,
 68        row_count: int = 0,
 69        column_count: int = 0,
 70        columns: Optional[List[str]] = None,
 71        original_filename: Optional[str] = None
 72    ) -> int:
 73        """Insert a new upload metadata record.
 74
 75        Records the shape and column names of an uploaded file. The
 76        actual row data is stored separately via `save_upload_data()`
 77        or atomically via `create_upload_with_data()`.
 78
 79        Args:
 80            filename: Stored/deduplicated filename on disk.
 81            file_type: File format identifier (e.g. "csv", "xlsx").
 82            row_count: Number of data rows in the file. Defaults to 0.
 83            column_count: Number of columns. Defaults to 0.
 84            columns: List of column header names. Stored as JSON.
 85            original_filename: Name of the file as uploaded by the user.
 86                Defaults to `filename` if not provided.
 87
 88        Returns:
 89            The `id` of the newly created upload row.
 90
 91        Raises:
 92            DatabaseError: On any database failure.
 93        """
 94        if original_filename is None:
 95            original_filename = filename
 96        try:
 97            with self.db.transaction() as conn:
 98                cursor = conn.cursor()
 99
100                columns_json = json.dumps(columns) if columns else None
101
102                cursor.execute('''
103                    INSERT INTO uploads (
104                        original_filename, filename, file_type, row_count, column_count, columns
105                    ) VALUES (?, ?, ?, ?, ?, ?)
106                ''', (
107                    original_filename,
108                    filename,
109                    file_type,
110                    row_count,
111                    column_count,
112                    columns_json
113                ))
114
115                upload_id = cursor.lastrowid
116                logger.info(f"Created upload {upload_id}: {filename}")
117                return upload_id
118
119        except sqlite3.IntegrityError as e:
120            logger.error(f"Integrity error creating upload: {e}")
121            raise DatabaseError(f"Failed to create upload: {e}") from e
122        except Exception as e:
123            logger.error(f"Failed to create upload: {e}")

230        file_type: Optional[str] = None,
231        row_count: Optional[int] = None,
232        column_count: Optional[int] = None,
233        columns: Optional[List[str]] = None
234    ) -> Optional[Dict[str, Any]]:
235        """Update one or more fields on an existing upload record.
236
237        Only non-None arguments are applied. Passing no updatable fields
238        is a no-op that returns the current row.
239
240        Args:
241            upload_id: Primary key of the upload to update.
242            original_filename: New original filename, if changing.
243            filename: New stored filename, if changing.
244            file_type: New file type, if changing.
245            row_count: New row count, if changing.
246            column_count: New column count, if changing.
247            columns: New column name list, if changing. Stored as JSON.
248
249        Returns:
250            The updated upload as a dict, or None if `upload_id` does
251            not exist.
252
253        Raises:
254            DatabaseError: On any database failure.
255        """
256        try:
257            with self.db.transaction() as conn:
258                cursor = conn.cursor()
259
260                updates = []
261                params = []
262
263                if original_filename is not None:
264                    updates.append('original_filename = ?')
265                    params.append(original_filename)
266
267                if filename is not None:
268                    updates.append('filename = ?')
269                    params.append(filename)
270
271                if file_type is not None:
272                    updates.append('file_type = ?')
273                    params.append(file_type)
274
275                if row_count is not None:
276                    updates.append('row_count = ?')
277                    params.append(row_count)
278
279                if column_count is not None:
280                    updates.append('column_count = ?')
281                    params.append(column_count)
282
283                if columns is not None:
284                    updates.append('columns = ?')
285                    params.append(json.dumps(columns))
286
287                if not updates:
288                    return self.get_upload_by_id(upload_id)
289
290                params.append(upload_id)
291
292                query = f"UPDATE uploads SET {', '.join(updates)} WHERE id = ?"
293                cursor.execute(query, params)
294
295                if cursor.rowcount == 0:
296                    logger.warning(f"Upload {upload_id} not found for update")
297                    return None
298
299            return self.get_upload_by_id(upload_id)
300
301        except Exception as e:
302            logger.error(f"Failed to update upload {upload_id}: {e}")
303            raise DatabaseError(f"Failed to update upload: {e}") from e

393    def save_upload_data(
394        self,
395        upload_id: int,
396        rows: List[Dict[str, Any]],
397        batch_size: int = 1000
398    ) -> int:
399        """Insert row data for an existing upload in batches.
400
401        Each row dict is JSON-serialized and stored with its zero-based
402        index. All batches are inserted within a single transaction, so
403        either all rows succeed or none do.
404
405        For atomic upload creation + data insertion, prefer
406        `create_upload_with_data()`.
407
408        Args:
409            upload_id: FK to the parent `uploads` row.
410            rows: List of row data dicts (one per source file row).
411            batch_size: Rows per `executemany` call. Affects memory
412                usage but not transactional behaviour. Defaults to 1000.
413
414        Returns:
415            Number of rows inserted.
416
417        Raises:
418            DatabaseError: If `upload_id` doesn't exist (FK violation),
419                a (upload_id, row_index) pair is duplicated, or on any
420                other database failure.
421        """
422        try:
423            total_inserted = 0
424
425            with self.db.transaction() as conn:
426                cursor = conn.cursor()
427
428                # Process in batches for better performance
429                for batch_start in range(0, len(rows), batch_size):
430                    batch = rows[batch_start:batch_start + batch_size]
431
432                    data = [
433                        (upload_id, idx + batch_start, json.dumps(row))
434                        for idx, row in enumerate(batch)
435                    ]
436
437                    cursor.executemany('''
438                        INSERT INTO upload_data (upload_id, row_index, row_data)
439                        VALUES (?, ?, ?)
440                    ''', data)
441
442                    total_inserted += len(batch)
443
444                logger.info(f"Saved {total_inserted} rows for upload {upload_id}")
445                return total_inserted
446
447        except sqlite3.IntegrityError as e:
448            logger.error(f"Integrity error saving upload data: {e}")
449            raise DatabaseError(f"Failed to save upload data: {e}") from e
450        except Exception as e:
451            logger.error(f"Failed to save upload data: {e}")
452            raise DatabaseError(f"Failed to save upload data: {e}") from e

649    def create_upload_with_data(
650        self,
651        filename: str,
652        file_type: str,
653        columns: List[str],
654        rows: List[Dict[str, Any]],
655        batch_size: int = 1000,
656        original_filename: Optional[str] = None
657    ) -> Dict[str, Any]:
658        """Create an upload record and insert all its data atomically.
659
660        Wraps the upload creation and data insertion in a single
661        transaction, so a failure at any point rolls back everything.
662        Preferred over calling `create_upload()` + `save_upload_data()`
663        separately.
664
665        Args:
666            filename: Stored/deduplicated filename on disk.
667            file_type: File format identifier (e.g. "csv", "xlsx").
668            columns: List of column header names.
669            rows: List of row data dicts. `row_count` and
670                `column_count` are derived automatically.
671            batch_size: Rows per `executemany` call. Defaults to 1000.
672            original_filename: Name as uploaded by the user. Defaults
673                to `filename` if not provided.
674
675        Returns:
676            Dict with `upload_id` and `rows_inserted`.
677
678        Raises:
679            DatabaseError: On any database failure (entire operation is
680                rolled back).
681        """
682        if original_filename is None:
683            original_filename = filename
684        try:
685            with self.db.transaction() as conn:
686                cursor = conn.cursor()
687
688                # Create upload record
689                columns_json = json.dumps(columns)
690                cursor.execute('''
691                    INSERT INTO uploads (
692                        original_filename, filename, file_type, row_count, column_count, columns
693                    ) VALUES (?, ?, ?, ?, ?, ?)
694                ''', (
695                    original_filename,
696                    filename,
697                    file_type,
698                    len(rows),
699                    len(columns),
700                    columns_json
701                ))
702
703                upload_id = cursor.lastrowid
704
705                # Insert data in batches
706                for batch_start in range(0, len(rows), batch_size):
707                    batch = rows[batch_start:batch_start + batch_size]
708
709                    data = [
710                        (upload_id, idx + batch_start, json.dumps(row))
711                        for idx, row in enumerate(batch)
712                    ]
713
714                    cursor.executemany('''
715                        INSERT INTO upload_data (upload_id, row_index, row_data)
716                        VALUES (?, ?, ?)
717                    ''', data)
718
719                logger.info(f"Created upload {upload_id} with {len(rows)} rows")
720

691                cursor.execute('''
692                    INSERT INTO uploads (
693                        original_filename, filename, file_type, row_count, column_count, columns
