src.database.schema

Database schema definition and creation for SQLite.

Defines every table, index, and trigger in the application database via the SchemaManager class. All DDL uses CREATE ... IF NOT EXISTS, so init_db() is safe to call on every startup.

Called automatically by connection.init_app() when create_tables=True.

Table dependency order (foreign keys flow downward):

receipts
categories
  └─ subcategories
       └─ types
            └─ parties
uploads
  └─ upload_data
accounts
transactions  ← references accounts, uploads, parties, receipts

243    def _create_receipts_table(self):
244        """Create the `receipts` table, indexes, and update trigger.
245
246        Stores metadata for uploaded receipt images. Receipts can
247        optionally be linked to a transaction via `transactions.receipt_id`.
248
249        Columns:
250            id: Auto-incrementing primary key.
251            original_filename: Name of the file as uploaded by the user.
252            stored_filename: Unique on-disk filename (deduplicated).
253            file_path: Full path to the stored file.
254            vendor: Extracted vendor/merchant name.
255            date: Extracted receipt date.
256            amount: Extracted total (must be >= 0).
257            confidence: Extraction confidence score (0–3).
258            selected_method: Which extraction method produced the result.
259            raw_text: Full OCR text output.
260            metadata: JSON string with additional extraction data.
261            created_at: Row creation timestamp.
262            updated_at: Auto-updated on row modification via trigger.
263
264        Indexes: vendor, date, created_at.
265        """
266        with self.db.get_connection() as conn:
267            cursor = conn.cursor()
268            cursor.execute('''
269                CREATE TABLE IF NOT EXISTS receipts (
270                    id INTEGER PRIMARY KEY AUTOINCREMENT,
271                    original_filename TEXT NOT NULL,
272                    stored_filename TEXT NOT NULL,
273                    file_path TEXT NOT NULL,
274                    vendor TEXT,
275                    date DATE,
276                    amount REAL CHECK(amount >= 0),
277                    confidence INTEGER DEFAULT 0 CHECK(confidence >= 0 AND confidence <= 3),
278                    selected_method TEXT,
279                    raw_text TEXT,
280                    metadata TEXT,
281                    created_at TIMESTAMP DEFAULT (strftime('%Y-%m-%d %H:%M:%f', 'now')),
282                    updated_at TIMESTAMP DEFAULT (strftime('%Y-%m-%d %H:%M:%f', 'now')),
283                    UNIQUE (stored_filename)
284                )
285            ''')
286
287            cursor.execute('''
288                CREATE INDEX IF NOT EXISTS idx_receipts_vendor 
289                ON receipts(vendor)
290            ''')
291            cursor.execute('''
292                CREATE INDEX IF NOT EXISTS idx_receipts_date 
293                ON receipts(date)
294            ''')
295            cursor.execute('''
296                CREATE INDEX IF NOT EXISTS idx_receipts_created 
297                ON receipts(created_at)
298            ''')
299
300            cursor.execute('''
301                CREATE TRIGGER IF NOT EXISTS update_receipts_timestamp 
302                AFTER UPDATE ON receipts
303                BEGIN
304                    UPDATE receipts SET updated_at = strftime('%Y-%m-%d %H:%M:%f', 'now')
305                    WHERE id = NEW.id;
306                END
307            ''')

410    def _create_transactions_table(self):
411        """Create the `transactions` table and indexes.
412
413        The central table — every imported bank/card transaction ends up
414        here. Links to the category hierarchy via `party_id` and
415        optionally to a receipt image via `receipt_id`.
416
417        Columns:
418            id: Auto-incrementing primary key.
419            transaction_date: Date the transaction occurred.
420            amount: Transaction value (positive for both debits/credits;
421                direction indicated by `is_credit`).
422            description: Raw description from the bank statement.
423            cleaned_description: Normalised description after parsing.
424            is_credit: 1 for income/credit, 0 for expense/debit.
425            is_kids: 1 if flagged as a child-related expense.
426            is_one_off: 1 if flagged as non-recurring.
427            account_id: FK → `accounts.id`. Restricts delete.
428            upload_id: FK → `uploads.id`. Restricts delete.
429            party_id: FK → `parties.id`. Restricts delete.
430            receipt_id: FK → `receipts.id`. Set to NULL on receipt delete.
431            created_at: Row creation timestamp.
432
433        Indexes: transaction_date, account_id, party_id, upload_id,
434            is_credit, receipt_id.
435        """
436        with self.db.get_connection() as conn:
437            cursor = conn.cursor()
438            cursor.execute('''
439                CREATE TABLE IF NOT EXISTS transactions (
440                    id INTEGER PRIMARY KEY AUTOINCREMENT,
441                    transaction_date DATE NOT NULL,
442                    amount REAL NOT NULL,
443                    description TEXT NOT NULL,
444                    cleaned_description TEXT,
445                    is_credit INTEGER NOT NULL CHECK(is_credit IN (0, 1)),
446                    is_kids INTEGER NOT NULL DEFAULT 0 CHECK(is_kids IN (0, 1)),
447                    is_one_off INTEGER NOT NULL DEFAULT 0 CHECK(is_one_off IN (0, 1)),
448                    account_id INTEGER,
449                    upload_id INTEGER NOT NULL,
450                    party_id INTEGER NOT NULL,
451                    receipt_id INTEGER,
452                    created_at TIMESTAMP DEFAULT (strftime('%Y-%m-%d %H:%M:%f', 'now')),
453                    FOREIGN KEY (account_id) REFERENCES accounts(id)
454                        ON DELETE RESTRICT ON UPDATE CASCADE,
455                    FOREIGN KEY (upload_id) REFERENCES uploads(id)
456                        ON DELETE RESTRICT ON UPDATE CASCADE,
457                    FOREIGN KEY (party_id) REFERENCES parties(id)
458                        ON DELETE RESTRICT ON UPDATE CASCADE,
459                    FOREIGN KEY (receipt_id) REFERENCES receipts(id)
460                        ON DELETE SET NULL ON UPDATE CASCADE
461                )
462            ''')
