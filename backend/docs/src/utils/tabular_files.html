src.utils.tabular_files

Tabular Data Processing Utility Module

A comprehensive utility for validating, previewing, and importing tabular data from various file formats.

Supported formats:
  • CSV, TSV, TXT (text-based)
  • XLS, XLSX, XLSM, XLSB (Excel)
  • ODS (OpenDocument)
  • Parquet
  • JSON
Example usage:

from tabular_processor import TabularProcessor

processor = TabularProcessor()

Validate a file

result = processor.validate("data.csv") print(result.to_json())

Preview data

preview = processor.preview("data.csv", num_rows=5) print(preview.to_json())

Import with options

data = processor.import_data( "data.xlsx", start_row=2, columns=["Name", "Age", "Email"], column_names=["full_name", "age_years", "email_address"] ) print(data.to_json())

 54    """Orchestrates validation, preview, and import of tabular files.
 55
 56    The public methods never raise for file-level problems (missing
 57    file, wrong format, read failure). Instead they return result
 58    objects with `is_valid=False` / `success=False` and a populated
 59    `errors` list. This keeps API routes simple — every code path
 60    produces a serializable response.
 61
 62    Attributes:
 63        default_encoding: Override encoding passed to the underlying
 64            `FileReader`. None for auto-detection.
 65        default_delimiter: Override delimiter for text files. None for
 66            auto-detection / file-type defaults.
 67        auto_detect_types: If True, run type inference on each column
 68            during validation. If False, all columns report "unknown".
 69        normalize_columns: If True, column names are normalised to
 70            valid identifiers (lowercase, underscores) during preview
 71            and import.
 72        reader: The underlying `FileReader` instance.

106    def _validate_file_path(self, file_path: Union[str, Path]) -> Path:
107        """Check that a path exists and points to a regular file.
108
109        Args:
110            file_path: Path to check.
111
112        Returns:
113            The path as a resolved `Path` object.
114
115        Raises:
116            FileNotFoundError: If the path doesn't exist or isn't a
117                file. Caught by the public methods and folded into
118                result-object errors.
119        """
120        path = Path(file_path)
121        if not path.exists():
122            raise FileNotFoundError(f"File not found: {path}")
123        if not path.is_file():
124            raise FileNotFoundError(f"Path is not a file: {path}")
125        return path

127    def _get_column_info(self, df: pd.DataFrame, sample_size: int = 5) -> List[ColumnInfo]:
128        """Build `ColumnInfo` objects for every column in a DataFrame.
129
130        Computes type (if `auto_detect_types`), nullability, null
131        count, unique count, and a small sample of non-null values
132        for each column.
133
134        Args:
135            df: The DataFrame to inspect.
136            sample_size: Number of sample values to capture per column.
137                Defaults to 5.
138
139        Returns:
140            List of `ColumnInfo`, one per column, in column order.
141        """
142        columns_info = []
143
144        for idx, col in enumerate(df.columns):
145            series = df[col]
146
147            non_null = series.dropna()
148            sample_values = non_null.head(sample_size).tolist()
149
150            col_info = ColumnInfo(
151                name=str(col),
152                index=int(idx),
153                data_type=infer_column_type(series) if self.auto_detect_types else "unknown",
154                nullable=bool(series.isna().any()),
155                sample_values=sample_values,
156                null_count=int(series.isna().sum()),
157                unique_count=int(series.nunique())
158            )
159            columns_info.append(col_info)
160
161        return columns_info

163    def validate(
164        self,
165        file_path: Union[str, Path],
166        min_rows: int = 0,
167        min_columns: int = 1,
168        required_columns: Optional[List[str]] = None,
169        sheet_name: Union[str, int] = 0
170    ) -> ValidationResult:
171        """Validate a tabular file and collect metadata about it.
172
173        Reads the entire file, checks it against the given constraints,
174        and returns a `ValidationResult` with detailed column info and
175        any errors or warnings.
176
177        Checks that produce errors (block import):
178            - File not found or not readable.
179            - Unsupported file type.
180            - Fewer than `min_rows` data rows.
181            - Fewer than `min_columns` columns.
182            - Any `required_columns` missing.
183
184        Checks that produce warnings (import proceeds):
185            - Duplicate column names.
186            - Completely empty columns.
187            - Completely empty rows.
188
189        Args:
190            file_path: Path to the file to validate.
191            min_rows: Minimum data rows required. Defaults to 0.
192            min_columns: Minimum columns required. Defaults to 1.
193            required_columns: Column names that must be present. If
194                None, no column-presence check is performed.
195            sheet_name: For Excel files, which sheet to validate.
196                Defaults to the first sheet.
197
198        Returns:
199            A `ValidationResult` with `is_valid` indicating whether
200            all checks passed, plus metadata, per-column info, and
201            populated `errors` / `warnings` lists.
202        """
203        errors = []
204        warnings = []
205
206        logger.debug(
207            f"Validating file: {Path(file_path).name} "
208            f"| min_rows={min_rows}, min_columns={min_columns}"
209        )
210
211        try:
212            path = self._validate_file_path(file_path)
213        except FileNotFoundError as e:
214            logger.warning(f"Validation failed — file not found: {file_path}")
215            return ValidationResult(
216                is_valid=False,
217                file_path=str(file_path),
218                file_name=Path(file_path).name,
219                file_type=FileType.UNKNOWN.value,
220                file_size_bytes=0,
221                row_count=0,
222                column_count=0,
223                columns=[],
224                has_header=True,
225                detected_delimiter=None,
226                detected_encoding="unknown",
227                errors=[str(e)],
228                warnings=[]
229            )
230
231        file_type = detect_file_type(path)
232        if file_type == FileType.UNKNOWN:
233            logger.warning(f"Unsupported file type: {path.suffix}")
234            return ValidationResult(
235                is_valid=False,
236                file_path=str(path),
237                file_name=path.name,
238                file_type=FileType.UNKNOWN.value,
239                file_size_bytes=get_file_size(path),
240                row_count=0,
241                column_count=0,
242                columns=[],
243                has_header=True,
244                detected_delimiter=None,
245                detected_encoding="unknown",
246                errors=[f"Unsupported file type: {path.suffix}"],
247                warnings=[]
248            )
249
250        file_size = get_file_size(path)
251        encoding = detect_encoding(path) if file_type in TEXT_TYPES else "N/A"
252        delimiter = detect_delimiter(path, encoding) if file_type in TEXT_TYPES else None
253
254        logger.debug(
255            f"File metadata: type={file_type.value}, size={file_size}, "
256            f"encoding={encoding}, delimiter={delimiter!r}"
257        )
258
259        try:
260            df = self.reader.read(
261                path,
262                file_type,
263                sheet_name=sheet_name
264            )
265
266            row_count = len(df)
267            column_count = len(df.columns)
268            columns_info = self._get_column_info(df)
269
270            if row_count < min_rows:
271                errors.append(
272                    f"File has {row_count} data rows, minimum required: {min_rows}"
273                )
274
275            if column_count < min_columns:
276                errors.append(
277                    f"File has {column_count} columns, minimum required: {min_columns}"
278                )
279
280            if required_columns:
281                actual_columns = set(str(c) for c in df.columns)
282                missing = set(required_columns) - actual_columns
283                if missing:
284                    errors.append(f"Missing required columns: {list(missing)}")
285
286            if df.columns.duplicated().any():
287                dup_cols = df.columns[df.columns.duplicated()].tolist()
288                warnings.append(f"Duplicate column names detected: {dup_cols}")
289
290            empty_cols = [col for col in df.columns if df[col].isna().all()]
291            if empty_cols:
292                warnings.append(f"Empty columns detected: {empty_cols}")
293
294            empty_rows = df.isna().all(axis=1).sum()
295            if empty_rows > 0:
296                warnings.append(f"File contains {empty_rows} completely empty rows")
297
298            is_valid = len(errors) == 0
299
300            if is_valid:
301                logger.debug(
302                    f"Validation passed for {path.name}: "
303                    f"{row_count} rows, {column_count} columns"
304                )
305            else:
306                logger.debug(
307                    f"Validation failed for {path.name}: {errors}"
308                )
309
310            if warnings:
311                logger.debug(f"Validation warnings for {path.name}: {warnings}")
312
313            return ValidationResult(
314                is_valid=is_valid,
315                file_path=str(path),
316                file_name=path.name,
317                file_type=file_type.value,
318                file_size_bytes=file_size,
319                row_count=row_count,
320                column_count=column_count,
321                columns=columns_info,
322                has_header=True,
323                detected_delimiter=delimiter,
324                detected_encoding=encoding,
325                errors=errors,
326                warnings=warnings
327            )
328
329        except Exception as e:
330            logger.error(f"Failed to read file for validation {path.name}: {e}")
331            return ValidationResult(
332                is_valid=False,
333                file_path=str(path),
334                file_name=path.name,
335                file_type=file_type.value,
336                file_size_bytes=file_size,
337                row_count=0,
338                column_count=0,
339                columns=[],
340                has_header=True,
341                detected_delimiter=delimiter,
342                detected_encoding=encoding,
343                errors=[f"Failed to read file: {str(e)}"],
344                warnings=[]
345            )

347    def preview(
348        self,
349        file_path: Union[str, Path],
350        num_rows: int = 10,
351        sheet_name: Union[str, int] = 0,
352        include_types: bool = True
353    ) -> PreviewResult:
354        """Generate a preview of the first N rows of a file.
355
356        Reads the full file (to get an accurate total row count), then
357        slices the first `num_rows` for the preview data. Column names
358        are normalised if `self.normalize_columns` is True.
359
360        Args:
361            file_path: Path to the file.
362            num_rows: Number of rows to include in the preview.
363                Defaults to 10.
364            sheet_name: For Excel files, which sheet to preview.
365            include_types: Whether to run type inference on preview
366                columns. Defaults to True.
367
368        Returns:
369            A `PreviewResult` with the preview data as JSON-safe row
370            dicts, plus column names, types, and total counts.
371            `success=False` with populated `errors` if the file can't
372            be read.
373        """
374        errors = []
375        warnings = []
376
377        logger.debug(f"Previewing file: {Path(file_path).name} | num_rows={num_rows}")
378
379        try:
380            path = self._validate_file_path(file_path)
381        except FileNotFoundError as e:
382            logger.warning(f"Preview failed — file not found: {file_path}")
383            return PreviewResult(
384                success=False,
385                file_name=Path(file_path).name,
386                file_type=FileType.UNKNOWN.value,
387                total_rows=0,
388                total_columns=0,
389                preview_row_count=0,
390                columns=[],
391                column_types={},
392                data=[],
393                errors=[str(e)]
394            )
395
396        file_type = detect_file_type(path)
397        if file_type == FileType.UNKNOWN:
398            logger.warning(f"Preview failed — unsupported file type: {path.suffix}")
399            return PreviewResult(
400                success=False,
401                file_name=path.name,
402                file_type=FileType.UNKNOWN.value,
403                total_rows=0,
404                total_columns=0,
405                preview_row_count=0,
406                columns=[],
407                column_types={},
408                data=[],
409                errors=[f"Unsupported file type: {path.suffix}"]
410            )
411
412        try:
413            df_full = self.reader.read(path, file_type, sheet_name=sheet_name)
414            total_rows = len(df_full)
415
416            df_preview = df_full.head(num_rows)
417
418            columns = [str(c) for c in df_preview.columns]
419            if self.normalize_columns:
420                columns = normalize_column_names(columns)
421                df_preview.columns = columns
422
423            column_types = {}
424            if include_types:
425                for col in df_preview.columns:
426                    column_types[str(col)] = infer_column_type(df_preview[col])
427
428            data = dataframe_to_json_records(df_preview)
429
430            logger.debug(
431                f"Preview generated for {path.name}: "
432                f"{total_rows} total rows, {len(columns)} columns, "
433                f"showing {len(df_preview)} rows"
434            )
435
436            return PreviewResult(
437                success=True,
438                file_name=path.name,
439                file_type=file_type.value,
440                total_rows=total_rows,
441                total_columns=len(columns),
442                preview_row_count=len(df_preview),
443                columns=columns,
444                column_types=column_types,
445                data=data,
446                errors=[],
447                warnings=warnings
448            )
449
450        except Exception as e:
451            logger.error(f"Failed to preview {path.name}: {e}")
452            return PreviewResult(
453                success=False,
454                file_name=path.name,
455                file_type=file_type.value,
456                total_rows=0,
457                total_columns=0,
458                preview_row_count=0,
459                columns=[],
460                column_types={},
461                data=[],
462                errors=[f"Failed to preview file: {str(e)}"]
463            )

465    def import_data(
466        self,
467        file_path: Union[str, Path],
468        start_row: int = 0,
469        columns: Optional[List[Union[int, str]]] = None,
470        column_names: Optional[List[str]] = None,
471        has_header: bool = True,
472        sheet_name: Union[str, int] = 0,
473        max_rows: Optional[int] = None,
474        skip_empty_rows: bool = True,
475        strip_whitespace: bool = True
476    ) -> ImportResult:
477        """Import data from a file with column selection and renaming.
478
479        Reads the file with the requested slicing, applies cleaning
480        transforms, optionally renames or normalises columns, and
481        returns the data as JSON-safe row dicts.
482
483        Args:
484            file_path: Path to the file.
485            start_row: Zero-based row index to start reading from.
486                Rows before this are skipped. Defaults to 0.
487            columns: Subset of columns to import, by name or index.
488                None for all columns.
489            column_names: Replacement column names. If the count
490                doesn't match, names are padded with `column_N` or
491                truncated, and a warning is added.
492            has_header: Whether the file has a header row. Defaults
493                to True.
494            sheet_name: For Excel files, which sheet to import.
495            max_rows: Maximum data rows to import. None for all.
496            skip_empty_rows: If True (default), drop rows where every
497                column is null.
498            strip_whitespace: If True (default), strip leading/trailing
499                whitespace from string values.
500
501        Returns:
502            An `ImportResult` with the imported data, counts, and
503            `column_mapping` dict showing original → new names if
504            renaming occurred. `success=False` with populated `errors`
505            if the import fails.
506        """
507        errors = []
508        warnings = []
509
510        logger.debug(
511            f"Importing file: {Path(file_path).name} "
512            f"| start_row={start_row}, has_header={has_header}, "
513            f"columns={columns}, max_rows={max_rows}"
514        )
515
516        try:
517            path = self._validate_file_path(file_path)
518        except FileNotFoundError as e:
519            logger.warning(f"Import failed — file not found: {file_path}")
520            return ImportResult(
521                success=False,
522                file_name=Path(file_path).name,
523                file_type=FileType.UNKNOWN.value,
524                start_row=start_row,
525                rows_imported=0,
526                rows_skipped=0,
527                columns_imported=[],
528                columns_requested=columns,
529                column_mapping=None,
530                data=[],
531                errors=[str(e)]
532            )
533
534        file_type = detect_file_type(path)
535        if file_type == FileType.UNKNOWN:
536            logger.warning(f"Import failed — unsupported file type: {path.suffix}")
537            return ImportResult(
538                success=False,
539                file_name=path.name,
540                file_type=FileType.UNKNOWN.value,
541                start_row=start_row,
542                rows_imported=0,
543                rows_skipped=0,
544                columns_imported=[],
545                columns_requested=columns,
546                column_mapping=None,
547                data=[],
548                errors=[f"Unsupported file type: {path.suffix}"]
549            )
550
551        try:
552            skiprows = None
553            if start_row > 0:
554                skiprows = list(range(0, start_row - 1))
555            header_setting = 0 if has_header else None
556
557            usecols = None
558            if columns is not None:
559                usecols = columns
560
561            df = self.reader.read(
562                path,
563                file_type,
564                nrows=max_rows,
565                skiprows=skiprows,
566                usecols=usecols,
567                header=header_setting,
568                sheet_name=sheet_name
569            )
570
571            rows_before = len(df)
572
573            if skip_empty_rows:
574                df = df.dropna(how='all')
575
576            rows_skipped = rows_before - len(df)
577
578            if rows_skipped > 0:
579                logger.debug(f"Skipped {rows_skipped} empty rows from {path.name}")
580
581            if strip_whitespace:
582                for col in df.select_dtypes(include=['object']).columns:
583                    df[col] = df[col].apply(
584                        lambda x: x.strip() if isinstance(x, str) else x
585                    )
586
587            # Apply custom column names
588            original_columns = [str(c) for c in df.columns]
589            column_mapping = None
590
591            if column_names is not None:
592                if len(column_names) != len(df.columns):
593                    msg = (
594                        f"Column names count ({len(column_names)}) doesn't match "
595                        f"data columns ({len(df.columns)}). Adjusting..."
596                    )
597                    warnings.append(msg)
598                    logger.warning(f"Import {path.name}: {msg}")
599
600                    if len(column_names) < len(df.columns):
601                        column_names = list(column_names) + [
602                            f"column_{i}" for i in range(len(column_names), len(df.columns))
603                        ]
604                    else:
605                        column_names = column_names[:len(df.columns)]
606
607                column_mapping = dict(zip(original_columns, column_names))
608                df.columns = column_names
609            elif self.normalize_columns:
610                new_columns = normalize_column_names(original_columns)
611                column_mapping = dict(zip(original_columns, new_columns))
612                df.columns = new_columns
613
614            final_columns = [str(c) for c in df.columns]
615            data = dataframe_to_json_records(df)
616
617            logger.debug(
618                f"Imported {path.name}: {len(df)} rows, "
619                f"{len(final_columns)} columns "
620                f"(skipped={rows_skipped + start_row})"
621            )
622
623            return ImportResult(
624                success=True,
