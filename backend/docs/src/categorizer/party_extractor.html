src.categorizer.party_extractor

Party name extraction from raw transaction descriptions.

Bank descriptions are noisy: "POS TRANSACTION TESCO METRO RATHMINES 12/03/24 REF:98765". This module strips the noise and extracts a stable party name ("TESCO METRO RATHMINES") that party_matcher can fuzzy-match against known merchants.

Two-stage pipeline, always called in order:

  1. clean() / clean_batch() — strip transaction-type prefixes, trailing dates, reference numbers, card masks, etc. Output is uppercase, whitespace-normalized.
  2. extract_party_name() / extract_party_names_batch() — drop stop words (LTD, DUBLIN, CARD, …) and take the first N remaining words.

Scalar methods exist for one-off use; batch methods are vectorised over pandas Series and are what the statement import pipeline actually calls.

 29class PartyExtractor:
 30    """
 31    Strips noise from transaction descriptions and extracts a party name.
 32
 33    Two-step API — **callers must clean before extracting**:
 34        cleaned = extractor.clean(description)
 35        party   = extractor.extract_party_name(cleaned)
 36
 37    Or for a whole statement:
 38        cleaned = extractor.clean_batch(descriptions)
 39        parties = extractor.extract_party_names_batch(cleaned)
 40
 41    Customization: pass `custom_patterns` / `custom_stop_words` to extend
 42    the defaults (they're appended, not replaced). All regexes are
 43    pre-compiled in `__init__` so per-row cost is just the match, not
 44    the compile.
 45    """
 46
 47    DEFAULT_PATTERNS = [
 48        # ── Leading transaction-type prefixes ──
 49        # Long form ("PAYMENT TO", "DIRECT DEBIT"), bank abbreviations
 50        # (POS, CNC, DD…), and short form ("PAYMENT", "TRANSFER").
 51        r'^(PAYMENT TO|TRANSFER TO|TRANSFER FROM|PURCHASE AT|POS TRANSACTION|'
 52        r'ONLINE PAYMENT|DIRECT DEBIT|DIRECT CREDIT|DEBIT CARD|CREDIT CARD)\s+',
 53        r'^(POS|CNC|TKN|DD|CT|VPP|INET|Rtd)\s+',
 54        r'^(PAYMENT|TRANSFER|PURCHASE|DEBIT|CREDIT)\s+',
 55
 56        # ── Trailing dates / times ──
 57        # Various formats banks append: dd/mm/yyyy, dd/mm hh:mm, dd-mm-yyyy, hh:mm
 58        r'\s+\d{2}/\d{2}/\d{2,4}.*$',
 59        r'\s+\d{2}/\d{2}\s+\d{1,2}:\d{2}.*$',
 60        r'\s+\d{2}/\d{2}\s+\d{1,2}$',
 61        r'\s+\d{2}-\d{2}-\d{2,4}.*$',
 62        r'\s+\d{1,2}:\d{2}.*$',
 63
 64        # ── Trailing reference / card / store codes ──
 65        # Long numbers, REF:xxx, masked cards (****1234), short codes, #123
 66        r'\s+\d{4,}$',
 67        r'\s+REF:.*$',
 68        r'\s+\*{4}\d{4}$',
 69        r'\s+\d{1,3}$',
 70        r'\s+[A-Z]\d{1,3}$',
 71        r'\s+#\d+.*$',
 72    ]
 73
 74    DEFAULT_STOP_WORDS = {
 75        # Articles / conjunctions
 76        'THE', 'AND', 'OR', 'FOR', 'WITH', 'AT', 'IN', 'ON', 'TO', 'FROM',
 77        # Transaction-type words that survived pattern stripping
 78        'PAYMENT', 'TRANSFER', 'TRANSACTION', 'PURCHASE', 'DEBIT', 'CREDIT',
 79        'ONLINE', 'DIRECT', 'WITHDRAWAL', 'DEPOSIT', 'FOREIGN',
 80        # Card networks / channels
 81        'CARD', 'VISA', 'MASTERCARD', 'AMEX', 'AMERICAN', 'EXPRESS',
 82        'POS', 'ATM', 'CNC', 'TKN', 'DD', 'CT', 'VPP', 'INET', 'RTD',
 83        # Generic merchant words
 84        'STORE', 'STORES', 'SHOP', 'CASH', 'FEE', 'FEES', 'CHARGE', 'CHARGES',
 85        'INTEREST', 'BRANCH', 'LOCATION', 'MERCHANT', 'SERVICE', 'SERVICES',
 86        # Corporate suffixes
 87        'LTD', 'LIMITED', 'LLC', 'INC', 'INCORPORATED', 'CORP', 'CORPORATION',
 88        'PTY', 'COMPANY', 'CO', 'GROUP', 'HOLDINGS',
 89        # Geography (don't want "TESCO DUBLIN" and "TESCO CORK" as different parties)
 90        'IRELAND', 'IRISH', 'DUBLIN', 'IE', 'IRL', 'UK', 'GB', 'USA', 'US',
 91        'AUSTRALIA', 'AUS', 'AU',
 92        # Day names
 93        'MON', 'TUE', 'WED', 'THU', 'FRI', 'SAT', 'SUN',
 94        'MONDAY', 'TUESDAY', 'WEDNESDAY', 'THURSDAY', 'FRIDAY',
 95        'SATURDAY', 'SUNDAY',
 96    }

 98    def __init__(
 99        self,
100        custom_patterns: Optional[List[str]] = None,
101        custom_stop_words: Optional[Set[str]] = None
102    ):
103        self.patterns = self.DEFAULT_PATTERNS + (custom_patterns or [])
104        self.stop_words = self.DEFAULT_STOP_WORDS.union(custom_stop_words or set())
105
106        # Pre-compile patterns once — avoids recompiling per row
107        self._compiled_patterns = [
108            re.compile(p, re.IGNORECASE) for p in self.patterns
109        ]
110        # Build single stop-word regex for vectorized removal
111        escaped = [re.escape(w) for w in sorted(self.stop_words, key=len, reverse=True)]
112        self._stop_word_pattern = re.compile(
113            r'\b(' + '|'.join(escaped) + r')\b',
114            re.IGNORECASE
115        )
116
117        logger.debug(
118            f"Initialized PartyExtractor: "
119            f"{len(self.patterns)} patterns, {len(self.stop_words)} stop words"
120        )
121
122    def clean(self, description: str) -> str:
123        """
124        Stage 1: strip noise from a raw description.
125
126        Applies every pattern in sequence, then removes punctuation and
127        standalone single letters, and normalizes whitespace. Output is
128        uppercase.
129
130        Returns an empty string for NaN/None input.
131        """
132        if pd.isna(description):
133            return ""
134        text = str(description).upper()
135        for pattern in self._compiled_patterns:
136            text = pattern.sub('', text)
137        text = re.sub(r'[^\w\s]', ' ', text)
138        text = re.sub(r'\s+[A-Z]\s+', ' ', text)
139        return ' '.join(text.split())
140
141    def extract_party_name(self, description: str, max_words: int = 3) -> str:
142        """
143        Stage 2: pull a party name from a **cleaned** description.
144
145        Drops stop words and single-character tokens, then takes the first
146        `max_words` survivors. If nothing survives, falls back to the first
147        30 characters of the input. If the input itself is empty, returns
148        "UNKNOWN".
149
150        Expects output from `clean()` — feeding it a raw description will
151        work but the fallback path will return raw junk.
152        """
153        if not description:
154            return "UNKNOWN"
155        words = description.split()
156        meaningful = [w for w in words if w not in self.stop_words and len(w) > 1]
157        if meaningful:
158            return ' '.join(meaningful[:max_words]).strip()
159        return description[:30] if len(description) > 30 else description
160
161    def clean_batch(self, descriptions: pd.Series) -> pd.Series:
162        """
163        Vectorized cleaning of an entire Series of descriptions.
164        
165        Uses pd.Series.str methods instead of Python-level loops.
166        """
167        logger.debug(f"Batch cleaning {len(descriptions)} descriptions")
168
169        s = descriptions.fillna('').astype(str).str.upper()
170
171        # Apply each compiled pattern across the whole series
172        for pattern in self._compiled_patterns:
173            s = s.str.replace(pattern, '', regex=True)
174
175        # Remove special characters
176        s = s.str.replace(r'[^\w\s]', ' ', regex=True)
177
178        # Remove standalone single letters
179        s = s.str.replace(r'\s+[A-Z]\s+', ' ', regex=True)
180
181        # Normalize whitespace
182        s = s.str.strip().str.replace(r'\s+', ' ', regex=True)
183
184        return s

186    def extract_party_names_batch(
187        self, descriptions: pd.Series, max_words: int = 3
188    ) -> pd.Series:
189        """
190        Stage 2: pull a party name from a **cleaned** description.
191
192        Drops stop words and single-character tokens, then takes the first
193        `max_words` survivors. If nothing survives, falls back to the first
194        30 characters of the input. If the input itself is empty, returns
195        "UNKNOWN".
196
197        Expects output from `clean()` — feeding it a raw description will
198        work but the fallback path will return raw junk.
199        """
200        logger.debug(f"Batch extracting party names from {len(descriptions)} descriptions")
201
202        # Remove stop words in one vectorized regex pass
203        filtered = descriptions.str.replace(
204            self._stop_word_pattern, '', regex=True
205        )
206        # Remove any resulting short (single-char) words
207        filtered = filtered.str.replace(r'\b\w\b', '', regex=True)
208        filtered = filtered.str.strip().str.replace(r'\s+', ' ', regex=True)
209
210        # Take first max_words words
211        # Split, slice, rejoin — this is the one part that's hard to avoid
212        extracted = filtered.apply(
213            lambda x: ' '.join(x.split()[:max_words]) if x else ''
214        )
215
216        # Fill empties: fall back to truncated original, then UNKNOWN
217        fallback = descriptions.str[:30]
218        empty_mask = extracted.str.strip() == ''
219        extracted = extracted.where(~empty_mask, fallback)
220        extracted = extracted.where(extracted.str.strip() != '', 'UNKNOWN')
221
222        return extracted
